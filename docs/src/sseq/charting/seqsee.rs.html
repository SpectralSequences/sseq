10/// A [`Backend`] that emits [SeqSee](https://github.com/JoeyBF/SeqSee) JSON.
11///
12/// SeqSee is a generic visualization tool for spectral sequences. It consumes a JSON file
13/// conforming to its [schema] and produces a self-contained, interactive HTML chart. This backend
14/// targets that schema so that the charts produced elsewhere in this crate can be rendered by
15/// SeqSee.
16///
17/// The document is assembled in memory and written out when the backend is dropped, since the JSON
18/// object must group all nodes together and all edges together, whereas the [`Backend`] callbacks
19/// interleave them.
20///
21/// [schema]: https://raw.githubusercontent.com/JoeyBF/SeqSee/refs/heads/master/seqsee/input_schema.json
22pub struct SeqSeeBackend<T: io::Write> {
23    out: T,
24    max: Bidegree,
25    /// Map from node id to its SeqSee node object.
26    nodes: Map<String, Value>,
27    /// The list of SeqSee edge objects.
28    edges: Vec<Value>,
29    /// The set of style names referenced by edges. Each becomes an attribute alias in the header so
30    /// that the edges referencing it validate against the schema.
31    styles: BTreeSet<String>,
32}
33
34impl<T: io::Write> SeqSeeBackend<T> {
35    pub fn new(out: T) -> Self {
36        Self {
37            out,
38            max: Bidegree::zero(),
39            nodes: Map::new(),
40            edges: Vec::new(),
41            styles: BTreeSet::new(),
42        }
43    }
44}
45
46/// Whether a style name denotes a differential, i.e. it is `d` followed by a page number.
47///
48/// Differentials are given a distinct color to match the other backends in this crate.
49fn is_differential(style: &str) -> bool {
50    style
51        .strip_prefix('d')
52        .is_some_and(|rest| !rest.is_empty() && rest.bytes().all(|b| b.is_ascii_digit()))
53}
54
55impl<T: io::Write> Backend for SeqSeeBackend<T> {
56    type Error = io::Error;
57
58    const EXT: &'static str = "json";
59
60    fn header(&mut self, max: Bidegree) -> Result<(), Self::Error> {
61        self.max = max;
62        Ok(())
63    }
64
65    // SeqSee draws its own grid based on the chart dimensions in the header, so there is nothing to
66    // emit for the grid lines.
67    fn line(&mut self, _start: Bidegree, _end: Bidegree, _style: &str) -> Result<(), Self::Error> {
68        Ok(())
69    }
70
71    // SeqSee draws its own axis labels, so there is nothing to emit here.
72    fn text(
73        &mut self,
74        _b: Bidegree,
75        _content: impl Display,
76        _orientation: Orientation,
77    ) -> Result<(), Self::Error> {
78        Ok(())
79    }
80
81    fn node(&mut self, b: Bidegree, n: usize) -> Result<(), Self::Error> {
82        if n == 0 || b.x() > self.max.x() || b.y() > self.max.y() {
83            return Ok(());
84        }
85
86        for k in 0..n {
87            // The `{:#}` (alternate) form of the `Display` impl is `(x,y,idx)`. Reusing it here and
88            // in `structline` keeps node ids and edge endpoints in sync.
89            let id = format!("{:#}", BidegreeGenerator::new(b, k));
90            self.nodes.insert(
91                id,
92                json!({
93                    "x": b.x(),
94                    "y": b.y(),
95                    "position": k,
96                }),
97            );
98        }
99        Ok(())
100    }
101
102    fn structline(
103        &mut self,
104        source: BidegreeGenerator,
105        target: BidegreeGenerator,
106        style: Option<&str>,
107    ) -> Result<(), Self::Error> {
108        if source.x() > self.max.x()
109            || source.y() > self.max.y()
110            || target.x() > self.max.x()
111            || target.y() > self.max.y()
112        {
113            return Ok(());
114        }
115
116        let mut edge = Map::new();
117        edge.insert("source".to_string(), Value::String(format!("{source:#}")));
118        edge.insert("target".to_string(), Value::String(format!("{target:#}")));
119        if let Some(style) = style {
120            self.styles.insert(style.to_string());
121            edge.insert("attributes".to_string(), json!([style]));
122        }
123        self.edges.push(Value::Object(edge));
124        Ok(())
125    }
126}
127
128impl<T: io::Write> Drop for SeqSeeBackend<T> {
129    fn drop(&mut self) {
130        // Register every referenced style as an attribute alias so that the edges validate.
131        let mut attributes = Map::new();
132        for style in &self.styles {
133            let attr = if is_differential(style) {
134                json!([{ "color": "blue" }])
135            } else {
136                json!([])
137            };
138            attributes.insert(style.clone(), attr);
139        }
140
141        let document = json!({
142            "header": {
143                "chart": {
144                    "width": { "min": 0, "max": self.max.x() },
145                    "height": { "min": 0, "max": self.max.y() },
146                },
147                "aliases": {
148                    "attributes": attributes,
149                },
150            },
151            "nodes": std::mem::take(&mut self.nodes),
152            "edges": std::mem::take(&mut self.edges),
153        });
154
155        let _ = serde_json::to_writer_pretty(&mut self.out, &document);
156        let _ = writeln!(self.out);
157    }
158}

164    #[test]
165    fn test_seqsee_output() {
166        let mut buf: Vec<u8> = Vec::new();
167        {
168            let mut backend = SeqSeeBackend::new(&mut buf);
169            // `init` draws the grid and axis labels via `line`/`text`, which are no-ops here.
170            backend.init(Bidegree::x_y(2, 2)).unwrap();
171
172            backend.node(Bidegree::x_y(0, 0), 1).unwrap();
173            backend.node(Bidegree::x_y(0, 1), 1).unwrap();
174            backend.node(Bidegree::x_y(1, 1), 1).unwrap();
175
176            // A multiplication by `h0` (a filtration-one product).
177            backend
178                .structline(
179                    BidegreeGenerator::new(Bidegree::x_y(0, 0), 0),
180                    BidegreeGenerator::new(Bidegree::x_y(0, 1), 0),
181                    Some("h0"),
182                )
183                .unwrap();
184            // A `d2` differential.
185            backend
186                .structline(
187                    BidegreeGenerator::new(Bidegree::x_y(1, 1), 0),
188                    BidegreeGenerator::new(Bidegree::x_y(0, 1), 0),
189                    Some("d2"),
190                )
191                .unwrap();
192            // Out of bounds: dropped silently.
193            backend.node(Bidegree::x_y(5, 5), 1).unwrap();
194        }
195
196        // Compare parsed values rather than the raw string so the assertion does not depend on
197        // JSON object key ordering, which varies with whether serde_json's `preserve_order`
198        // feature is enabled by feature unification in the surrounding build.
199        let produced: Value = serde_json::from_slice(&buf).unwrap();
200        let expected = json!({
201            "header": {
202                "chart": {
203                    "width": { "min": 0, "max": 2 },
204                    "height": { "min": 0, "max": 2 },
205                },
206                "aliases": {
207                    "attributes": {
208                        "h0": [],
209                        "d2": [{ "color": "blue" }],
210                    },
211                },
212            },
213            "nodes": {
214                "(0,0,0)": { "x": 0, "y": 0, "position": 0 },
215                "(0,1,0)": { "x": 0, "y": 1, "position": 0 },
216                "(1,1,0)": { "x": 1, "y": 1, "position": 0 },
217            },
218            "edges": [
219                { "source": "(0,0,0)", "target": "(0,1,0)", "attributes": ["h0"] },
220                { "source": "(1,1,0)", "target": "(0,1,0)", "attributes": ["d2"] },
221            ],
222        });
223        assert_eq!(produced, expected);
224    }
