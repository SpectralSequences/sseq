31const TILE_M: usize = MW * MSTRIPS; // output rows per CTA
32const TILE_K: usize = TK;
33const KL: usize = TILE_K / 64;
34const THREADS: u32 = (2 * THREADS_PER_WG) as u32; // producer warpgroup + consumer warpgroup
35const NG: u32 = (NB / 64) as u32; // output column-limbs per CTA

37/// A `CUtensorMap` passed by value as a (grid-constant) kernel argument.
38///
39/// `repr(transparent)` so the pointer cudarc's typed launch builder pushes is the address of the
40/// 128-byte descriptor itself.
41#[repr(transparent)]
42struct TmaArg(sys::CUtensorMap);
43
44// SAFETY: `DeviceRepr` requires the type to be plain data that is valid to memcpy into a kernel
45// parameter. `CUtensorMap` is an opaque 128-byte POD descriptor with no host pointers or padding
46// invariants, and the kernel declares the matching parameter `const __grid_constant__ CUtensorMap`.

49/// A CUDA device with the `matmul_b1` kernel loaded, ready to launch.
50pub struct GpuContext {
51    ctx: Arc<CudaContext>,
52    /// Per-thread CUDA streams for *this* context, created lazily (see [`Self::stream`]). Owned by
53    /// the context so a thread using several contexts (e.g. one per device) gets a distinct stream
54    /// per context. The mutex is held only for the map lookup, never across a GPU submission, so it
55    /// does not serialize device work — unlike the whole-op lock this design replaced.
56    streams: Mutex<HashMap<ThreadId, Arc<CudaStream>>>,
57    #[allow(dead_code)]
58    module: Arc<CudaModule>,
59    kernel: CudaFunction,
60}
61
62impl GpuContext {
63    /// Open device `device_id` and load the kernel onto it.
64    ///
65    /// Fails if there is no usable device, or if the embedded PTX is the stub `build.rs` emits when
66    /// `nvcc` is absent, in which case the kernel is missing from the module.
67    pub fn new(device_id: usize) -> anyhow::Result<Self> {
68        let ctx = CudaContext::new(device_id)?;
69        let ptx = Ptx::from_src(String::from_utf8(PTX_IMAGE.to_vec())?);
70        let module = ctx.load_module(ptx)?;
71        let kernel = module.load_function("matmul_b1_kernel")?;
72        Ok(Self {
73            ctx,
74            streams: Mutex::new(HashMap::new()),
75            module,
76            kernel,
77        })
78    }
79
80    /// The device's compute capability as `(major, minor)`. The kernel requires 9.0 (Hopper).
81    pub fn compute_capability(&self) -> anyhow::Result<(i32, i32)> {
82        let major = self.ctx.attribute(
83            sys::CUdevice_attribute_enum::CU_DEVICE_ATTRIBUTE_COMPUTE_CAPABILITY_MAJOR,
84        )?;
85        let minor = self.ctx.attribute(
86            sys::CUdevice_attribute_enum::CU_DEVICE_ATTRIBUTE_COMPUTE_CAPABILITY_MINOR,
87        )?;
88        Ok((major, minor))
89    }
90
91    pub fn default_stream(&self) -> Arc<CudaStream> {
92        self.ctx.default_stream()
93    }
94
95    /// A CUDA stream **private to the calling OS thread**.
96    ///
97    /// Created lazily on first use and reused thereafter, cached per thread in this context's
98    /// `streams` map. Submitting through this instead of the context's single `default_stream()`
99    /// lets calls from different threads run on distinct streams — overlapping transfers and
100    /// kernels instead of serializing — while all sub-steps of one call share one stream, which
101    /// keeps ordering correct within a thread. This is what lets `try_mul` (and the row-reduce) run
102    /// lock-free from many threads at once.
103    pub fn stream(&self) -> Arc<CudaStream> {
104        self.streams
105            .lock()
106            .expect("stream cache poisoned")
107            .entry(std::thread::current().id())
108            // Fall back to the context default stream if creation fails (e.g. a poisoned context),
109            // so the caller sees a normal Err and degrades to CPU rather than panicking.
110            .or_insert_with(|| {
111                self.ctx
112                    .new_stream()
113                    .unwrap_or_else(|_| self.ctx.default_stream())
114            })
115            .clone()
116    }
117
118    pub fn kernel(&self) -> &CudaFunction {
119        &self.kernel
120    }
121}

123/// Multiply two bit-packed F₂ matrices on the GPU.
124///
125/// Operands are plain **row-major, K-major** limb arrays — the exact layout `fp::Matrix::to_bytes`
126/// produces (little-endian `u64` limbs, one bit per entry, `columns.div_ceil(64)` limbs per row, no
127/// inter-row padding):
128///
129/// - `a`: the `m`×`k` left operand, `m * k.div_ceil(64)` limbs.
130/// - `b`: the `k`×`n` right operand, `k * n.div_ceil(64)` limbs.
131///
132/// Returns C = A·B as `m * n.div_ceil(64)` limbs in the same layout, ready to hand to
133/// `fp::Matrix::from_data`.
134pub fn matmul_b1_raw(
135    gpu: &GpuContext,
136    a: &[u64],
137    m: usize,
138    k: usize,
139    b: &[u64],
140    n: usize,
141) -> anyhow::Result<Vec<u64>> {
142    Ok(matmul_b1_inner(gpu, a, m, k, b, n, 1)?.0)
143}
144
145/// Like [`matmul_b1_raw`], but also returns the average **kernel-only** wall time in seconds.
146///
147/// The time is averaged over `time_iters` back-to-back launches, and excludes host
148/// (de)serialization, the TMA-layout pre-arrangement, and the H2D/D2H copies.
149///
150/// The kernel zeroes its SMEM accumulator and writes C with a bulk-tensor *store* (overwrite, not
151/// accumulate), so repeated launches against the same device buffers are idempotent and the
152/// returned limbs are the correct product.
153pub fn matmul_b1_raw_timed(
154    gpu: &GpuContext,
155    a: &[u64],
156    m: usize,
157    k: usize,
158    b: &[u64],
159    n: usize,
160    time_iters: usize,
161) -> anyhow::Result<(Vec<u64>, f64)> {
162    matmul_b1_inner(gpu, a, m, k, b, n, time_iters.max(1))
163}

165#[allow(clippy::too_many_arguments)]
166fn matmul_b1_inner(
167    gpu: &GpuContext,
168    a: &[u64],
169    m: usize,
170    k: usize,
171    b: &[u64],
172    n: usize,
173    time_iters: usize,
174) -> anyhow::Result<(Vec<u64>, f64)> {
175    let n_lim = n.div_ceil(64);
176    assert_eq!(a.len(), m * k.div_ceil(64), "A limb count mismatch");
177    assert_eq!(b.len(), k * n_lim, "B limb count mismatch");
178
179    let k_padded = k.next_multiple_of(TILE_K);
180    // Pad M to a whole number of M-tiles; the extra padded rows produce zeros
181    // that the `take(m)` readback trims.
182    let m_padded = m.next_multiple_of(TILE_M);
183    let m_tiles = m_padded / TILE_M;
184    let k_chunks = k_padded / TILE_K;
185    // Each CTA computes a TILE_M×(NG*64) output block via MSTRIPS m64n128 wgmmas,
186    // so B (and the C output) are grouped/padded to whole NG-limb column tiles.
187    let n_groups = n_lim.div_ceil(NG as usize);
188    let n_padded_lim = n_groups * NG as usize;
189
190    let stream = gpu.stream();
191
192    let a_padded = pad_2d(a, m, k.div_ceil(64), m_padded, k_padded / 64);
193    let b_padded = pad_2d(b, k, n_lim, k_padded, n_lim);
194
195    let a_interleaved = interleave_a(&a_padded, m_padded, k_padded);
196    let bt = transpose_b(&b_padded, k_padded, n_lim);
197
198    let a_dev = stream.clone_htod(&a_interleaved)?;
199    let bt_dev = stream.clone_htod(&bt)?;
200    let c_dev = stream.alloc_zeros::<u64>(m_padded * n_padded_lim)?;
201
202    // Raw device addresses for the TMA descriptors. The returned guards keep the
203    // reads ordered on the stream; hold them until after the launch.
204    let (a_ptr, _ga) = a_dev.device_ptr(&stream);
205    let (b_ptr, _gb) = bt_dev.device_ptr(&stream);
206    let (c_ptr, _gc) = c_dev.device_ptr(&stream);
207
208    let tma_a = encode_tma(
209        a_ptr,
210        [32, (k_chunks * m_tiles * TILE_M) as u64],
211        [32, TILE_M as u32],
212        128,
213        sys::CUtensorMapSwizzle_enum::CU_TENSOR_MAP_SWIZZLE_128B,
214    )?;
215    let tma_b = encode_tma(
216        b_ptr,
217        [32, (k_chunks * n_groups * NG as usize * 64) as u64],
218        [32, (NG as usize * 64) as u32],
219        128,
220        sys::CUtensorMapSwizzle_enum::CU_TENSOR_MAP_SWIZZLE_128B,
221    )?;
222    let tma_c = encode_tma(
223        c_ptr,
224        [(n_padded_lim * 2) as u64, m_padded as u64],
225        [(NG as usize * 2) as u32, TILE_M as u32],
226        (n_padded_lim * 8) as u64,
227        sys::CUtensorMapSwizzle_enum::CU_TENSOR_MAP_SWIZZLE_NONE,
228    )?;
229
230    // Dynamic SMEM per CTA: sA + sB + 2*sC (double-buffered) + 2*STAGES mbarriers.
231    let tile_a = TILE_M * KL; // TILE_M-row A block
232    let tile_b = NG as usize * 64 * KL; // (NG*64)-col B tile
233    let smem_u64 = STAGES * tile_a + STAGES * tile_b + 2 * NG as usize * TILE_M + 2 * STAGES;
234    let smem_bytes = (smem_u64 * std::mem::size_of::<u64>()) as u32;
235
236    // Opt in to >48 KB shared memory (Hopper static default cap).
237    gpu.kernel.set_attribute(
238        sys::CUfunction_attribute_enum::CU_FUNC_ATTRIBUTE_MAX_DYNAMIC_SHARED_SIZE_BYTES,
239        smem_bytes as i32,
240    )?;
241
242    // Persistent grid: co-resident CTAs = (occupancy per SM) × SM count, so the grid exactly fills
243    // the machine and the kernel's persistent loop sweeps all output tiles in grouped-rasterized
244    // order. The CTAs are independent, so the grid size is a throughput parameter only: any size is
245    // correct, and fewer CTAs simply do more tile-iterations each.
246    let sms = gpu
247        .ctx
248        .attribute(sys::CUdevice_attribute_enum::CU_DEVICE_ATTRIBUTE_MULTIPROCESSOR_COUNT)?
249        as u32;
250    let occ = gpu
251        .kernel
252        .occupancy_max_active_blocks_per_multiprocessor(THREADS, smem_bytes as usize, None)?
253        .max(1);
254    let num_ctas = (occ * sms).max(1);
255    if std::env::var("FP_CUDA_DEBUG").is_ok() {
256        eprintln!("[fp-cuda] occ={occ}/SM sms={sms} num_ctas={num_ctas} smem={smem_bytes}B");
257    }
258
259    let ta = TmaArg(tma_a);
260    let tb = TmaArg(tma_b);
261    let tc = TmaArg(tma_c);
262    let mt = m_tiles as u32;
263    let ng = n_groups as u32;
264    let m_val = m_padded as u32;
265    let k_val = k_padded as u32;
266
267    let launch = || -> Result<(), cudarc::driver::DriverError> {
268        let cfg = LaunchConfig {
269            grid_dim: (num_ctas, 1, 1),
270            block_dim: (THREADS, 1, 1),
271            shared_mem_bytes: smem_bytes,
272        };
273        let mut lb = stream.launch_builder(&gpu.kernel);
274        lb.arg(&ta)
275            .arg(&tb)
276            .arg(&tc)
277            .arg(&mt)
278            .arg(&ng)
279            .arg(&m_val)
280            .arg(&k_val);
281        // SAFETY: the seven pushed arguments match `matmul_b1_kernel`'s parameter list in order and
282        // type, `smem_bytes` is the size the kernel was granted above via
283        // MAX_DYNAMIC_SHARED_SIZE_BYTES, and the device buffers behind the tensor maps outlive this
284        // closure (their guards are held until after the final synchronize).
285        unsafe { lb.launch(cfg) }?;
286        Ok(())
287    };
288
289    // Warm up once (untimed) when measuring, so the timed loop excludes any first-launch
290    // JIT/allocation costs.
291    if time_iters > 1 {
292        launch()?;
293        stream.synchronize()?;
294    }
295
296    let start = Instant::now();
297    for _ in 0..time_iters {
298        launch()?;
299    }
300    stream.synchronize()?;
301    let kernel_secs = start.elapsed().as_secs_f64() / time_iters as f64;
302
303    let c_all = stream.clone_dtoh(&c_dev)?;
304    let c_limbs: Vec<u64> = c_all
305        .chunks_exact(n_padded_lim)
306        .take(m)
307        .flat_map(|row| row[..n_lim].iter().copied())
308        .collect();
309    Ok((c_limbs, kernel_secs))
310}

312/// Encode a 2D row-major TMA tensor map of UINT32 elements.
313fn encode_tma(
314    dev_ptr: sys::CUdeviceptr,
315    gdim: [u64; 2],
316    boxdim: [u32; 2],
317    row_stride_bytes: u64,
318    swizzle: sys::CUtensorMapSwizzle_enum,
319) -> anyhow::Result<sys::CUtensorMap> {
320    let gstride = [row_stride_bytes];
321    let elemstride = [1u32, 1u32];
322    let mut tmap = MaybeUninit::<sys::CUtensorMap>::uninit();
323    // SAFETY: `gdim`, `gstride`, `boxdim` and `elemstride` are the rank-2 arrays the driver reads
324    // (rank is passed as 2, and `gstride` is rank-1 by the API's contract of omitting the innermost
325    // dimension). `dev_ptr` is a live device allocation owned by the caller. On success the driver
326    // has fully written `tmap`, so the `assume_init` below is sound; on failure we return early.
327    unsafe {
328        sys::cuTensorMapEncodeTiled(
329            tmap.as_mut_ptr(),
330            sys::CUtensorMapDataType_enum::CU_TENSOR_MAP_DATA_TYPE_UINT32,
331            2,
332            dev_ptr as *mut c_void,
333            gdim.as_ptr(),
334            gstride.as_ptr(),
335            boxdim.as_ptr(),
336            elemstride.as_ptr(),
337            sys::CUtensorMapInterleave_enum::CU_TENSOR_MAP_INTERLEAVE_NONE,
338            swizzle,
339            sys::CUtensorMapL2promotion_enum::CU_TENSOR_MAP_L2_PROMOTION_NONE,
340            sys::CUtensorMapFloatOOBfill_enum::CU_TENSOR_MAP_FLOAT_OOB_FILL_NONE,
341        )
342        .result()?;
343        Ok(tmap.assume_init())
344    }
345}

347/// Gather A into plain row-major K-major tiles for TMA 128B swizzle.
348///
349/// Output: contiguous tiles, each TILE_M rows × KL u64s, so a row is 128 bytes — the swizzle width.
350/// The TMA applies the 128B swizzle on load, so the host layout is the natural row-major sub-block:
351/// tile row `row` holds K bits `kk*TILE_K .. +TILE_K` of global row `bi*TILE_M + row`, zero-padded
352/// out of bounds.
353///
354/// Tiles are ordered: for K-chunk kk=0..k_chunks-1, then M-tile bi=0..m_tiles-1.
355fn interleave_a(a: &[u64], m: usize, k: usize) -> Vec<u64> {
356    let sa = k / 64;
357    let k_chunks = k / TILE_K;
358    let m_tiles = m / TILE_M;
359    let tile_u64s = TILE_M * KL;
360    let mut out = vec![0u64; k_chunks * m_tiles * tile_u64s];
361
362    for kk in 0..k_chunks {
363        for bi in 0..m_tiles {
364            let base = (kk * m_tiles + bi) * tile_u64s;
365            for row in 0..TILE_M {
366                for kl in 0..KL {
367                    let global_row = bi * TILE_M + row;
368                    let global_kl = kk * KL + kl;
369                    let val = if global_row < m && global_kl < sa {
370                        a[global_row * sa + global_kl]
371                    } else {
372                        0
373                    };
374                    out[base + row * KL + kl] = val;
375                }
376            }
377        }
378    }
379    out
380}

382/// Pre-transpose B into plain row-major K-major tiles for TMA 128B swizzle.
383///
384/// Each (k_chunk, column group) tile is NB = NG*64 rows (= the NG*64 output columns of the group) ×
385/// KL u64s (= TILE_K K bits); the consumer feeds it to MSTRIPS m64n128 wgmmas that share it.
386/// Operand row `lg*64 + jj` is output column `cg*NG*64 + lg*64 + jj`; element `[..][kl] bit` is bit
387/// `jj` of `B[k_chunk*TILE_K + kl*64 + bit][cg*NG + lg]`.
388///
389/// Groups whose limb runs past `n_lim` are left zero-padded. Output is row-major; the TMA applies
390/// the swizzle on load.
391fn transpose_b(b: &[u64], k: usize, n_lim: usize) -> Vec<u64> {
392    let k_chunks = k / TILE_K;
393    let ng = NG as usize;
394    let n_groups = n_lim.div_ceil(ng);
395    let tile = ng * 64 * KL; // NB rows × KL u64
396    let mut out = vec![0u64; k_chunks * n_groups * tile];
397    let mut buf = [0u64; TILE_K];
398
399    for kk in 0..k_chunks {
400        for cg in 0..n_groups {
401            let base = (kk * n_groups + cg) * tile;
402            for lg in 0..ng {
403                let limb = cg * ng + lg;
404                if limb >= n_lim {
405                    continue; // padded column group → leave zeros
406                }
407                for (i, slot) in buf.iter_mut().enumerate() {
408                    let br = kk * TILE_K + i;
409                    *slot = if br < k { b[br * n_lim + limb] } else { 0 };
410                }
411                for jj in 0..64usize {
412                    let j = lg * 64 + jj; // operand row within the NB-col tile
413                    for kl in 0..KL {
414                        let mut val: u64 = 0;
415                        for bit in 0..64usize {
416                            val |= ((buf[kl * 64 + bit] >> jj) & 1) << bit;
417                        }
418                        out[base + j * KL + kl] = val;
419                    }
420                }
421            }
422        }
423    }
424    out
425}

427fn pad_2d(src: &[u64], rows: usize, stride: usize, nr: usize, ns: usize) -> Vec<u64> {
428    if rows == nr && stride == ns {
429        return src.to_vec();
430    }
431    let mut out = vec![0u64; nr * ns];
432    for r in 0..rows {
433        let n = stride.min(ns);
434        out[r * ns..r * ns + n].copy_from_slice(&src[r * stride..r * stride + n]);
435    }
436    out
437}

445    /// `true` iff a usable CUDA device is present, probed once for the whole test binary.
446    ///
447    /// `GpuContext::new` initializes the CUDA driver through cudarc, which *panics* (rather than
448    /// returning `Err`) when no driver library is present — as on GPU-less CI. We silence the panic
449    /// hook and catch the unwind so the probe reports "no GPU" instead of aborting the run. Every
450    /// device-touching test gates on this and returns early when it is `false`, so the whole GPU
451    /// test suite is disabled cleanly (rather than failing) wherever there is no device.
452    fn gpu_available() -> bool {
453        static AVAIL: OnceLock<bool> = OnceLock::new();
454        *AVAIL.get_or_init(|| {
455            let prev_hook = std::panic::take_hook();
456            std::panic::set_hook(Box::new(|_| {}));
457            let ok = std::panic::catch_unwind(|| GpuContext::new(0).is_ok()).unwrap_or(false);
458            std::panic::set_hook(prev_hook);
459            ok
460        })
461    }

463    /// Regression for the per-thread stream cache being scoped to its context.
464    ///
465    /// A single thread that builds two `GpuContext`s must get two *distinct* streams — the cache
466    /// was once keyed by thread alone, so the second context silently reused the first's stream
467    /// (wrong context/device). The same context must still reuse its own cached stream. Uses device
468    /// 0 twice: distinct instances, so distinct streams, without needing a second GPU.
469    #[test]
470    fn stream_is_scoped_per_context() {
471        if !gpu_available() {
472            return; // no usable GPU/driver in this environment — nothing to exercise
473        }
474        let a = GpuContext::new(0).expect("GPU is available");
475        let b = GpuContext::new(0).expect("GPU is available");
476        let (sa, sb) = (a.stream(), b.stream());
477        assert!(
478            !Arc::ptr_eq(&sa, &sb),
479            "distinct GpuContexts must not share a per-thread stream"
480        );
481        assert!(
482            Arc::ptr_eq(&sa, &a.stream()),
483            "a context must reuse its own cached stream"
484        );
485        assert!(
486            Arc::ptr_eq(&sb, &b.stream()),
487            "a context must reuse its own cached stream"
488        );
489    }
