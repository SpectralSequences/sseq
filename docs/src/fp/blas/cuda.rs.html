1//! GPU dispatch for F₂ matrix multiplication (Hopper `wgmma.b1`).
2
3use std::sync::OnceLock;
4
5use fp_cuda::GpuContext;
6
7use crate::{matrix::Matrix, prime::TWO};
8
9/// Smallest `min(m, k, n)` for which we attempt the GPU.
10///
11/// Below this the host marshalling (bit-repack into TMA tiles + copies) costs more than it saves.
12const DEFAULT_THRESHOLD: usize = 2048;
13
14/// The threshold in use, overridable via the `FP_CUDA_THRESHOLD` environment variable.
15fn threshold() -> usize {
16    std::env::var("FP_CUDA_THRESHOLD")
17        .ok()
18        .and_then(|v| v.parse().ok())
19        .unwrap_or(DEFAULT_THRESHOLD)
20}
21
22/// The process-wide GPU context, created lazily on first use.
23///
24/// `None` if no usable device is present (no driver, no Hopper GPU, or the kernel PTX is the
25/// nvcc-absent build stub), or if `FP_CUDA_DISABLE` is set.
26///
27/// Shared as `&'static` with no lock: `GpuContext` is `Send + Sync`, every submission goes through
28/// a per-thread stream ([`GpuContext::stream`]) so concurrent callers overlap instead of
29/// serializing, and device buffers are per-call, so there is no shared state to guard.
30fn context() -> Option<&'static GpuContext> {
31    static GPU: OnceLock<Option<GpuContext>> = OnceLock::new();
32    GPU.get_or_init(|| {
33        if std::env::var_os("FP_CUDA_DISABLE").is_some() {
34            return None;
35        }
36        GpuContext::new(0).ok()
37    })
38    .as_ref()
39}
40
41/// Row-major, K-major `u64` limbs — the exact layout `fp_cuda::matmul_b1_raw` expects.
42///
43/// That is `rows × columns.div_ceil(64)` limbs with no inter-row padding. Uses `Matrix::to_bytes`,
44/// which already strips the physical row stride.
45fn to_limbs(m: &Matrix) -> Vec<u64> {
46    let stride = m.columns().div_ceil(64);
47    let mut bytes = Vec::with_capacity(m.rows() * stride * 8);
48    m.to_bytes(&mut bytes).expect("Vec writes never fail");
49    let (chunks, _) = bytes.as_chunks::<8>();
50    chunks.iter().map(|&c| u64::from_le_bytes(c)).collect()
51}
52
53/// Try to compute `a · b` on the GPU.
54///
55/// This is consulted by `<&Matrix as Mul>::mul` before the CPU BLAS path: for large enough
56/// `p = 2` products it converts the operands to the raw row-major limb layout `fp-cuda` expects,
57/// runs the kernel, and rebuilds a [`Matrix`]. Anything that makes the GPU path unavailable or
58/// unsuitable — no device, a launch error, or a below-threshold size — returns `None`.
59///
60/// Assumes `a.prime() == b.prime() == 2` and `a.columns() == b.rows()`.
61pub(super) fn try_mul(a: &Matrix, b: &Matrix) -> Option<Matrix> {
62    debug_assert_eq!(a.prime(), TWO);
63    debug_assert_eq!(b.prime(), TWO);
64    debug_assert_eq!(a.columns(), b.rows());
65
66    let (m, k, n) = (a.rows(), a.columns(), b.columns());
67    let t = threshold();
68    if m < t || k < t || n < t {
69        return None;
70    }
71
72    let ctx = context()?;
73    let a_limbs = to_limbs(a);
74    let b_limbs = to_limbs(b);
75
76    // Lock-free: `matmul_b1_raw` submits on the calling thread's own stream with per-call device
77    // buffers, so concurrent callers do not interfere (see [`context`]).
78    let c = fp_cuda::matmul_b1_raw(ctx, &a_limbs, m, k, &b_limbs, n).ok()?;
79    Some(Matrix::from_data(TWO, m, n, c))
80}
