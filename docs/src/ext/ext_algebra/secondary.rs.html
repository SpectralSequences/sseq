23/// The Adams $d_2$ as an [`ExtDifferential`], with shift $(n, s) \mapsto (n-1, s+2)$.
24///
25/// Its matrix out of a bidegree is the $d_2$ recorded by the secondary resolution's homotopies, so
26/// attaching it to an [`ExtModule`] makes [`ExtModule::cohomology_subquotient`] the $E_3$ page.
27pub(crate) struct SecondaryCoboundary<CC: FreeChainComplex>
28where
29    CC::Algebra: PairAlgebra,
30{
31    res_lift: Arc<SecondaryResolution<CC>>,
32}
33
34impl<CC: FreeChainComplex> ExtDifferential for SecondaryCoboundary<CC>
35where
36    CC::Algebra: PairAlgebra,
37{
38    fn shift(&self) -> Bidegree {
39        Bidegree::n_s(-1, 2)
40    }
41
42    fn matrix(&self, b: Bidegree) -> Option<Matrix> {
43        let res = self.res_lift.underlying();
44        let p = res.prime();
45        let target = b + self.shift();
46
47        // Off the first quadrant Ext vanishes, a known zero. Inside it, an unresolved end means the
48        // page there is unknown, so there is no differential: treating an unresolved target as
49        // zero would make every source generator look like a surviving cycle.
50        let gens = |x: Bidegree| -> Option<usize> {
51            if x.n() < 0 || x.s() < 0 {
52                Some(0)
53            } else if res.has_computed_bidegree(x) {
54                Some(res.number_of_gens_in_bidegree(x))
55            } else {
56                None
57            }
58        };
59        let rows = gens(b)?;
60        let cols = gens(target)?;
61
62        let mut mat = Matrix::new(p, rows, cols);
63        // `m[i]` is the d2 of the i-th generator of `b`, as a vector at `target`.
64        if rows > 0 && cols > 0 {
65            let m = self.res_lift.homotopy(b.s() + 2).homotopies.hom_k(b.t());
66            if !m.is_empty() && !m[0].is_empty() {
67                for (i, row) in m.iter().enumerate() {
68                    for (k, &v) in row.iter().enumerate() {
69                        if v != 0 {
70                            mat.row_mut(i).set_entry(k, v);
71                        }
72                    }
73                }
74            }
75        }
76        Some(mat)
77    }
78}

80/// A single secondary product `x · y` in $\Mod_{C\lambda^2}$, where `y` is an $E_3$-surviving
81/// class. See [`SecondaryExtAlgebra::secondary_multiply_into`].
82pub struct SecondaryProduct {
83    /// The multiplicand: an $E_3$-surviving generator of the unit at the queried bidegree `b`.
84    pub source: BidegreeElement,
85    /// The $\Ext$ part of the product, in bidegree `b + x.degree()`.
86    pub ext_part: FpVector,
87    /// The $\lambda$ part of the product, in bidegree `b + x.degree() + LAMBDA_BIDEGREE`, already
88    /// reduced by the image of $d_2$.
89    pub lambda_part: FpVector,
90}

92/// The secondary layer over an [`ExtModule`]: the $d_2$ differential and the $\Mod_{C\lambda^2}$
93/// product.
94///
95/// This wraps [`SecondaryResolution`] and [`SecondaryResolutionHomomorphism`] for `M` and the unit
96/// `k`. It is a separate type from [`ExtModule`] because it requires `CC::Algebra: PairAlgebra`, a
97/// bound the primary layer does not impose.
98pub struct SecondaryExtAlgebra<CC: FreeChainComplex>
99where
100    CC::Algebra: PairAlgebra,
101{
102    module: Arc<ExtModule<CC>>,
103    res_lift: Arc<SecondaryResolution<CC>>,
104    /// `Arc`-shared with `res_lift` when `M == k`.
105    unit_lift: Arc<SecondaryResolution<CC>>,
106    /// The module with [`SecondaryCoboundary`] attached, so that its
107    /// [`cohomology_subquotient`](ExtModule::cohomology_subquotient) is the $E_3$ page.
108    alg_d2: ExtModule<CC>,
109    /// The unit Ext with $d_2$ attached: the $E_3$ page of $\Ext(k, k)$.
110    unit_d2: ExtModule<CC>,
111    /// Secondary lift of the multiplication map, cached per multiplier class `(degree, coords)`.
112    secondary_products: DashMap<BidegreeElement, Arc<SecondaryResolutionHomomorphism<CC, CC>>>,
113}
114
115impl<CC: FreeChainComplex + 'static> SecondaryExtAlgebra<CC>
116where
117    CC::Algebra: PairAlgebra,
118{
119    /// Build the secondary layer over `module`.
120    ///
121    /// Construction is cheap; call [`extend_all`](Self::extend_all) to compute the secondary
122    /// resolutions and $E_3$ pages.
123    pub fn new(module: Arc<ExtModule<CC>>) -> Self {
124        let res_lift = Arc::new(SecondaryResolution::new(Arc::clone(module.resolution())));
125        let unit_lift = if module.is_unit() {
126            Arc::clone(&res_lift)
127        } else {
128            Arc::new(SecondaryResolution::new(Arc::clone(
129                module.algebra().resolution(),
130            )))
131        };
132        // The coboundary reads the secondary homotopies lazily, so this is cheap before `extend_all`.
133        let alg_d2 = ExtModule::intrinsic(Arc::clone(module.resolution())).with_differential(
134            Arc::new(SecondaryCoboundary {
135                res_lift: Arc::clone(&res_lift),
136            }),
137        );
138        let unit_d2 = ExtModule::intrinsic(Arc::clone(module.algebra().resolution()))
139            .with_differential(Arc::new(SecondaryCoboundary {
140                res_lift: Arc::clone(&unit_lift),
141            }));
142        Self {
143            module,
144            res_lift,
145            unit_lift,
146            alg_d2,
147            unit_d2,
148            secondary_products: DashMap::new(),
149        }
150    }
151
152    /// Extend the secondary resolutions as far as the underlying resolutions allow.
153    /// Must be called before [`d2`](Self::d2), [`page_data`](Self::page_data) or
154    /// [`secondary_multiply_into`](Self::secondary_multiply_into); the $E_3$ pages are
155    /// then computed on demand from the extended homotopies.
156    pub fn extend_all(&self) {
157        self.res_lift.extend_all();
158        if !self.module.is_unit() {
159            self.unit_lift.extend_all();
160        }
161    }
162
163    /// Sharding entry point: compute only the secondary resolution data for filtration `s`,
164    /// distributed across machines sharing a save directory (see the `secondary` example docs).
165    /// Mirrors [`SecondaryLift::compute_partial`]. Returns before any $E_3$ page is built.
166    pub fn compute_partial(&self, s: i32) {
167        self.res_lift.compute_partial(s);
168        if !self.module.is_unit() {
169            self.unit_lift.compute_partial(s);
170        }
171    }
172
173    /// The primary [`ExtModule`] this is built on.
174    pub fn module(&self) -> &Arc<ExtModule<CC>> {
175        &self.module
176    }
177
178    /// The prime of the underlying resolution.
179    fn prime(&self) -> fp::prime::ValidPrime {
180        self.module.prime()
181    }
182
183    /// The secondary differential $d_2(x)$, a class in bidegree `(n - 1, s + 2)`.
184    ///
185    /// Returns `None` if the target bidegree has not been computed (so $d_2$ is unknown). A
186    /// computed-but-zero differential is `Some` of a zero class.
187    pub fn d2(&self, x: &BidegreeElement) -> Option<BidegreeElement> {
188        let b = x.degree();
189        let target = b + Bidegree::n_s(-1, 2);
190        let res = self.res_lift.underlying();
191        if !(b.t() > 0 && res.has_computed_bidegree(target)) {
192            return None;
193        }
194
195        let target_dim = res.number_of_gens_in_bidegree(target);
196        let mut out = FpVector::new(self.prime(), target_dim);
197
198        // `m[i]` is the d2 of the i-th generator of `b`, as a vector at `target`. This is exactly
199        // the matrix `SecondaryResolution::e3_page` reads to install d2 differentials.
200        let m = self.res_lift.homotopy(b.s() + 2).homotopies.hom_k(b.t());
201        if !m.is_empty() && !m[0].is_empty() {
202            let p = self.prime().as_u32();
203            for (i, c) in x.vec().iter_nonzero() {
204                for (k, &v) in m[i].iter().enumerate() {
205                    out.add_basis_element(k, (c * v) % p);
206                }
207            }
208        }
209        Some(BidegreeElement::new(target, out))
210    }
211
212    /// Whether `x` is a $d_2$-cycle (a permanent class through $E_3$).
213    pub fn survives(&self, x: &BidegreeElement) -> Option<bool> {
214        self.d2(x).map(|d| d.vec().is_zero())
215    }
216
217    /// The $E_3$-page subquotient of $\Ext(M, k)$ at bidegree `b` — the cohomology of
218    /// the primary Ext with the Adams $d_2$ attached, on the shared
219    /// [`cohomology_subquotient`](ExtModule::cohomology_subquotient) path.
220    pub fn page_data(&self, b: Bidegree) -> Subquotient {
221        self.alg_d2
222            .cohomology_subquotient(b)
223            .expect("call extend_all() first (and query a computed bidegree)")
224    }
225
226    /// The $E_3$-page subquotient of the unit $\Ext(k, k)$ at bidegree `b`.
227    pub fn unit_page_data(&self, b: Bidegree) -> Subquotient {
228        self.unit_d2
229            .cohomology_subquotient(b)
230            .expect("call extend_all() first (and query a computed bidegree)")
231    }
232}

234impl<CC: FreeChainComplex + crate::chain_complex::AugmentedChainComplex + 'static>
235    SecondaryExtAlgebra<CC>
236where
237    CC::Algebra: PairAlgebra,
238{
239    /// The secondary lift of multiplication by `x`, built and cached per multiplier class. The
240    /// returned lift is *not* extended; [`secondary_multiply_into`](Self::secondary_multiply_into)
241    /// extends it as needed. Exposed so callers can drive sharded computation
242    /// (`lift.underlying().extend_all()` then `lift.compute_partial(s)`).
243    pub fn secondary_product_lift(
244        &self,
245        x: &BidegreeElement,
246    ) -> Arc<SecondaryResolutionHomomorphism<CC, CC>> {
247        if let Some(map) = self.secondary_products.get(x) {
248            return Arc::clone(&map);
249        }
250
251        let name = format!("prod_{x}",);
252        let underlying = Arc::new(ResolutionHomomorphism::from_class(
253            name,
254            Arc::clone(self.module.resolution()),
255            Arc::clone(self.module.algebra().resolution()),
256            x.degree(),
257            &x.vec().iter().collect::<Vec<_>>(),
258        ));
259        let lift = Arc::new(SecondaryResolutionHomomorphism::new(
260            Arc::clone(&self.res_lift),
261            Arc::clone(&self.unit_lift),
262            underlying,
263        ));
264
265        Arc::clone(
266            self.secondary_products
267                .entry(x.clone())
268                .or_insert(lift)
269                .value(),
270        )
271    }
272
273    /// The secondary product of `x` with every $E_3$-surviving class of the unit at bidegree `b`,
274    /// computed in $\Mod_{C\lambda^2}$.
275    ///
276    /// Returns one [`SecondaryProduct`] per surviving generator at `b`; the $\lambda$ part is
277    /// already reduced by the image of $d_2$. The caller must have run [`extend_all`](Self::extend_all)
278    /// and computed both resolutions far enough.
279    pub fn secondary_multiply_into(
280        &self,
281        x: &BidegreeElement,
282        b: Bidegree,
283    ) -> Vec<SecondaryProduct> {
284        let p = self.prime();
285        let shift = x.degree();
286        // `hom_k` queries the page at the λ-part's source, which only it knows.
287        let lambda_page = |bd: Bidegree| self.alg_d2.cohomology_subquotient(bd);
288
289        let ext_dim = self
290            .module
291            .resolution()
292            .number_of_gens_in_bidegree(b + shift);
293        let lambda_dim = self
294            .module
295            .resolution()
296            .number_of_gens_in_bidegree(b + shift + LAMBDA_BIDEGREE);
297
298        let page = self.unit_page_data(b);
299        let n = page.subspace_dimension();
300        if n == 0 {
301            return Vec::new();
302        }
303
304        let lift = self.secondary_product_lift(x);
305        lift.underlying().extend_all();
306        lift.extend_all();
307
308        let mut outputs = vec![FpVector::new(p, ext_dim + lambda_dim); n];
309        lift.hom_k(
310            Some(&lambda_page),
311            b,
312            page.subspace_gens(),
313            outputs.iter_mut().map(FpVector::as_slice_mut),
314        );
315
316        page.subspace_gens()
317            .zip(outputs)
318            .map(|(g, out)| SecondaryProduct {
319                source: BidegreeElement::new(b, g.to_owned()),
320                ext_part: out.slice(0, ext_dim).to_owned(),
321                lambda_part: out.slice(ext_dim, ext_dim + lambda_dim).to_owned(),
322            })
323            .collect()
324    }
325}

334    #[test]
335    fn test_sphere_d2() {
336        let res = Arc::new(construct_standard::<false, _, _>("S_2", None).unwrap());
337        // Far enough to reach the first Adams differential d2(h4) = h0 h3^2 at (14, 3).
338        res.compute_through_stem(Bidegree::n_s(16, 6));
339        let e2 = Arc::new(ExtModule::intrinsic(res));
340        let sec_e2 = SecondaryExtAlgebra::new(Arc::clone(&e2));
341        sec_e2.extend_all();
342
343        // h_0, h_1, h_2 are permanent cycles.
344        for (n, s) in [(0, 1), (1, 1), (3, 1)] {
345            let h = e2.generator(BidegreeGenerator::new(Bidegree::n_s(n, s), 0));
346            let h_survives = sec_e2
347                .survives(&h)
348                .unwrap_or_else(|| panic!("h at (n={n}, s={s}) should have a computed d2"));
349            assert!(h_survives, "h at (n={n}, s={s}) should survive d2");
350            let h_d2 = sec_e2
351                .d2(&h)
352                .unwrap_or_else(|| panic!("h at (n={n}, s={s}) should have a computed d2"));
353            assert!(
354                h_d2.vec().is_zero(),
355                "d2 of a permanent class should vanish"
356            );
357        }
358
359        // The first Adams differential: d2(h4) = h0 h3^2, the generator of Ext^{3,17} at (14, 3).
360        let h4 = e2.generator(BidegreeGenerator::new(Bidegree::n_s(15, 1), 0));
361        let d = sec_e2.d2(&h4).expect("d2(h4) target should be computed");
362        assert_eq!(d.degree(), Bidegree::n_s(14, 3));
363        assert_eq!(e2.dimension(Bidegree::n_s(14, 3)), 1);
364        assert!(!d.vec().is_zero(), "d2(h4) = h0 h3^2 should be nonzero");
365        let h4_survives = sec_e2.survives(&h4).expect("h4 should have a computed d2");
366        assert!(!h4_survives, "h4 should not survive d2");
367    }

369    #[test]
370    fn d2_as_ext_differential_reproduces_the_e3_page() {
371        // `cohomology_subquotient` under `SecondaryCoboundary` must match `page_data`: same
372        // dimension at every bidegree and the same d2-image quotient.
373        let res = Arc::new(construct_standard::<false, _, _>("S_2", None).unwrap());
374        res.compute_through_stem(Bidegree::n_s(16, 6));
375        let e2 = Arc::new(ExtModule::intrinsic(Arc::clone(&res)));
376        let sec = SecondaryExtAlgebra::new(Arc::clone(&e2));
377        sec.extend_all();
378
379        let coboundary = Arc::new(SecondaryCoboundary {
380            res_lift: Arc::clone(&sec.res_lift),
381        });
382        let e2_d2 = ExtModule::intrinsic(Arc::clone(&res)).with_differential(coboundary);
383
384        let mut saw_nontrivial = false;
385        for n in 0..=15 {
386            for s in 1..=5 {
387                let b = Bidegree::n_s(n, s);
388                let Some(dim) = e2_d2.cohomology_dimension(b) else {
389                    continue;
390                };
391                let page = sec.page_data(b);
392                assert_eq!(
393                    dim,
394                    page.dimension(),
395                    "E3 dimension mismatch at (n={n}, s={s})"
396                );
397                // The subquotient's denominator is the d2-image: reducing any E2 vector
398                // by it must agree with the spectral sequence's page quotient.
399                let sq = e2_d2.cohomology_subquotient(b).unwrap();
400                assert_eq!(
401                    sq.dimension(),
402                    page.dimension(),
403                    "E3 subquotient dimension mismatch at (n={n}, s={s})"
404                );
405                if page.dimension() != e2.dimension(b) {
406                    saw_nontrivial = true; // d2 actually killed something here
407                }
408            }
409        }
410        assert!(
411            saw_nontrivial,
412            "expected d2 to be nontrivial somewhere in range (e.g. h4 at (15,1) → (14,3))"
413        );
414    }

416    /// At the top of the computed region the incoming $d_2$ is unknown, so no page is claimed.
417    ///
418    /// $d_2$ shifts $(n, s) \mapsto (n-1, s+2)$, so what lands on `b` comes from $(n+1, s-2)$ —
419    /// past the last computed stem here. Reading that as rank zero would report the $E_2$
420    /// dimension as though nothing could hit `b`, at exactly the bidegree where something might.
421    #[test]
422    fn an_unknown_incoming_d2_is_not_read_as_zero() {
423        let res = Arc::new(construct_standard::<false, _, _>("S_2", None).unwrap());
424        res.compute_through_stem(Bidegree::n_s(8, 4));
425        let e2 = Arc::new(ExtModule::intrinsic(Arc::clone(&res)));
426        let sec = SecondaryExtAlgebra::new(Arc::clone(&e2));
427        sec.extend_all();
428        let e2_d2 = ExtModule::intrinsic(Arc::clone(&res)).with_differential(Arc::new(
429            SecondaryCoboundary {
430                res_lift: Arc::clone(&sec.res_lift),
431            },
432        ));
433
434        let b = Bidegree::n_s(8, 2);
435        assert!(res.has_computed_bidegree(b), "b itself must be computed");
436        assert!(
437            !res.has_computed_bidegree(Bidegree::n_s(9, 0)),
438            "the incoming source must be unresolved for this to be the edge case"
439        );
440        assert_eq!(e2_d2.cohomology_dimension(b), None);
441        assert!(e2_d2.cohomology_subquotient(b).is_none());
442
443        // An *empty* bidegree is a different matter: nothing can survive there, so the unknown
444        // incoming differential does not make the answer unknown.
445        let empty = Bidegree::n_s(8, 4);
446        assert_eq!(e2.dimension(empty), 0);
447        assert!(!res.has_computed_bidegree(Bidegree::n_s(9, 2)));
448        assert_eq!(e2_d2.cohomology_dimension(empty), Some(0));
449        assert_eq!(
450            e2_d2.cohomology_subquotient(empty).map(|q| q.dimension()),
451            Some(0)
452        );
453    }

455    #[test]
456    fn secondary_product_runs_and_ext_part_is_the_primary_product() {
457        // Every product's Ext part equals the primary Ext product x · source.
458        let res = Arc::new(construct_standard::<false, _, _>("S_2", None).unwrap());
459        res.compute_through_stem(Bidegree::n_s(10, 8));
460        let e2 = Arc::new(ExtModule::intrinsic(Arc::clone(&res)));
461        let sec = SecondaryExtAlgebra::new(Arc::clone(&e2));
462        sec.extend_all();
463
464        // Multiply h0 into the classes at (0,1); the lone survivor is h0, so the Ext
465        // part must be the primary product h0 · h0 = h0².
466        let h0 = e2.generator(BidegreeGenerator::new(Bidegree::n_s(0, 1), 0));
467        let products = sec.secondary_multiply_into(&h0, Bidegree::n_s(0, 1));
468        assert!(
469            !products.is_empty(),
470            "expected a secondary product at (0,1)"
471        );
472        for prod in &products {
473            let primary = e2.multiply(&h0, &prod.source);
474            assert_eq!(
475                prod.ext_part,
476                primary.vec().to_owned(),
477                "secondary product Ext part must equal the primary product"
478            );
479        }
480    }
