19/// The ring $\Ext(k, k)$, backed by a resolution of the base field `k`.
20///
21/// A product is realised by a [`ResolutionHomomorphism`] from a fixed multiplier class, which
22/// computes the products of that class with all of $\Ext(k, k)$ at once. One such map is cached per
23/// generator; a product by a general class is assembled from them on request (see
24/// [`class_product_map`](Self::class_product_map)). Products are computed up to a sign.
25pub struct ExtAlgebra<CC: FreeChainComplex> {
26    /// Resolution of the base field `k`. Ring products live here.
27    resolution: Arc<CC>,
28    /// One multiplication map per generator of $\Ext(k, k)$, `res(k) → res(k)`, built on demand.
29    products: DashMap<BidegreeGenerator, Arc<ResolutionHomomorphism<CC, CC>>>,
30}
31
32impl<CC: FreeChainComplex> ExtAlgebra<CC> {
33    /// Build the ring $\Ext(k, k)$ from a resolution of `k`.
34    pub fn new(resolution: Arc<CC>) -> Self {
35        Self {
36            resolution,
37            products: DashMap::new(),
38        }
39    }
40
41    /// The resolution of `k` backing this ring.
42    pub fn resolution(&self) -> &Arc<CC> {
43        &self.resolution
44    }
45
46    /// The prime of the underlying resolution.
47    pub fn prime(&self) -> ValidPrime {
48        self.resolution.prime()
49    }
50
51    /// The dimension of $\Ext^{s,t}(k, k)$ at the given bidegree.
52    pub fn dimension(&self, b: Bidegree) -> usize {
53        self.resolution.number_of_gens_in_bidegree(b)
54    }
55
56    /// The basis generators of $\Ext(k, k)$ at the given bidegree.
57    pub fn basis(&self, b: Bidegree) -> Vec<BidegreeGenerator> {
58        (0..self.dimension(b))
59            .map(|i| BidegreeGenerator::new(b, i))
60            .collect()
61    }
62
63    /// A class in $\Ext(k, k)$ from its coordinates in the generator basis at bidegree `b`.
64    pub fn element(&self, b: Bidegree, coords: &[u32]) -> BidegreeElement {
65        assert_eq!(self.dimension(b), coords.len());
66        BidegreeElement::new(b, FpVector::from_slice(self.prime(), coords))
67    }
68
69    /// A single generator of $\Ext(k, k)$ as a class.
70    pub fn generator(&self, g: BidegreeGenerator) -> BidegreeElement {
71        let ambient = self.dimension(g.degree());
72        assert!(ambient > g.idx());
73        g.into_element(self.prime(), ambient)
74    }
75}

77impl<CC> ExtAlgebra<CC>
78where
79    CC: FreeChainComplex + AugmentedChainComplex,
80{
81    /// The multiplication map for a single generator `g` of $\Ext(k, k)$ (`res(k) → res(k)`), built
82    /// and cached on first use. The returned map is *not* guaranteed to be extended.
83    pub fn generator_product_map(
84        &self,
85        g: BidegreeGenerator,
86    ) -> Arc<ResolutionHomomorphism<CC, CC>> {
87        cached_generator_product_map(&self.products, &self.resolution, &self.resolution, g)
88    }
89
90    /// The multiply-by-`x` chain self-map of `res(k)` (`res(k) → res(k)`), extended through `max`.
91    ///
92    /// A generator with coefficient one returns the cached
93    /// [`generator_product_map`](Self::generator_product_map) itself; any other nonzero class adds
94    /// the cached generator maps via [`ResolutionHomomorphism::linear_combination`] (no
95    /// quasi-inverse lift).
96    pub fn class_product_map(
97        &self,
98        x: &BidegreeElement,
99        max: Bidegree,
100    ) -> Arc<ResolutionHomomorphism<CC, CC>> {
101        let summands: Vec<(u32, Arc<ResolutionHomomorphism<CC, CC>>)> = x
102            .vec()
103            .iter_nonzero()
104            .map(|(idx, c)| {
105                let map = self.generator_product_map(BidegreeGenerator::new(x.degree(), idx));
106                map.extend_through_stem(max);
107                (c, map)
108            })
109            .collect();
110        match summands.as_slice() {
111            [(1, map)] => Arc::clone(map),
112            [] => {
113                // With no initial images, extending lifts zero everywhere.
114                let hom = Arc::new(ResolutionHomomorphism::new(
115                    String::new(),
116                    Arc::clone(&self.resolution),
117                    Arc::clone(&self.resolution),
118                    x.degree(),
119                ));
120                hom.extend_through_stem(max);
121                hom
122            }
123            _ => Arc::new(ResolutionHomomorphism::linear_combination(
124                String::new(),
125                &summands,
126                max,
127            )),
128        }
129    }
130
131    /// Left-multiplication by `x ∈ Ext(k, k)`, applied to every basis generator of $\Ext(k, k)$ at
132    /// bidegree `b`. See [`ExtModule::multiply_into`] for the return convention.
133    pub fn multiply_into(&self, x: &BidegreeElement, b: Bidegree) -> Option<Matrix> {
134        products_into(
135            &self.resolution,
136            &self.resolution,
137            &self.products,
138            self.prime(),
139            x,
140            b,
141        )
142    }
143
144    /// The ring product `x · y` (both in $\Ext(k, k)$) if it lies in the computed range, else
145    /// `None`. The result lies in bidegree `x.degree() + y.degree()`.
146    pub fn try_multiply(
147        &self,
148        x: &BidegreeElement,
149        y: &BidegreeElement,
150    ) -> Option<BidegreeElement> {
151        let matrix = self.multiply_into(x, y.degree())?;
152        Some(combine_product(
153            &matrix,
154            y,
155            x.degree() + y.degree(),
156            self.prime(),
157        ))
158    }
159
160    /// The ring product `x · y`, both in $\Ext(k, k)$. Panics if out of the computed range; use
161    /// [`try_multiply`](Self::try_multiply) to handle that case.
162    pub fn multiply(&self, x: &BidegreeElement, y: &BidegreeElement) -> BidegreeElement {
163        self.try_multiply(x, y).expect(
164            "multiply: product is out of the computed range; compute further or use try_multiply",
165        )
166    }
167}

169/// The module $\Ext(M, k)$ over the ring [`ExtAlgebra`] $\Ext(k, k)$, backed by a resolution of
170/// `M`.
171///
172/// Products follow the [`ExtAlgebra`] conventions, with the multiplier in $\Ext(M, k)$ and the
173/// chain maps running `res(M) → res(k)`. When `M == k` the module and ring share one resolution
174/// (see [`is_unit`](Self::is_unit)).
175pub struct ExtModule<CC: FreeChainComplex> {
176    /// Resolution of `M`; the module's classes and the module-action products land in its Ext.
177    resolution: Arc<CC>,
178    /// Shared handle to the ring $\Ext(k, k)$. `Arc`-shared so all modules over the same `k` reuse
179    /// one ring cache.
180    algebra: Arc<ExtAlgebra<CC>>,
181    /// One multiplication map per generator of $\Ext(M, k)$, `res(M) → res(k)`, built on demand.
182    /// Read through [`product_cache`](Self::product_cache).
183    products: DashMap<BidegreeGenerator, Arc<ResolutionHomomorphism<CC, CC>>>,
184}

207        if !self.is_unit() {
208            self.resolution.compute_through_stem(max);
209        }
210    }

293    /// The per-generator product maps for this module.
294    ///
295    /// When `M == k` these are the ring's maps, so the module and its ring build each one once.
296    fn product_cache(&self) -> &DashMap<BidegreeGenerator, Arc<ResolutionHomomorphism<CC, CC>>> {
297        if self.is_unit() {
298            &self.algebra.products
299        } else {
300            &self.products
301        }
302    }
303
304    /// The multiplication map for a single generator `g` of $\Ext(M, k)$ (`res(M) → res(k)`), built
305    /// and cached on first use. The returned map is *not* guaranteed to be extended;
306    /// [`multiply_into`](Self::multiply_into) extends it as needed.
307    pub fn generator_product_map(
308        &self,
309        g: BidegreeGenerator,
310    ) -> Arc<ResolutionHomomorphism<CC, CC>> {
311        cached_generator_product_map(
312            self.product_cache(),
313            &self.resolution,
314            self.algebra.resolution(),
315            g,
316        )
317    }
318
319    /// Left-multiplication by the class `x` (in $\Ext(M, k)$), applied to every basis generator of
320    /// $\Ext(k, k)$ at bidegree `b`.
321    ///
322    /// Returns `None` when the product is out of the computed range — that is, when `b` or
323    /// `b + x.degree()` has not been resolved — so callers never mistake an uncomputed product for a
324    /// zero one. Otherwise returns a matrix with one row per generator of $\Ext(k, k)$ at `b`; row
325    /// `j` is the product `x · g_j` expressed in the generator basis of $\Ext(M, k)$ at bidegree
326    /// `b + x.degree()`. A computed-but-empty bidegree yields a valid zero-dimension matrix, not
327    /// `None`.
328    pub fn multiply_into(&self, x: &BidegreeElement, b: Bidegree) -> Option<Matrix> {
329        products_into(
330            &self.resolution,
331            self.algebra.resolution(),
332            self.product_cache(),
333            self.prime(),
334            x,
335            b,
336        )
337    }
338
339    /// The product `x · y` if it lies in the computed range, else `None`. See
340    /// [`multiply_into`](Self::multiply_into) for the operand conventions (`x ∈ Ext(M, k)`, `y ∈
341    /// Ext(k, k)`). The result lies in bidegree `x.degree() + y.degree()`.
342    pub fn try_multiply(
343        &self,
344        x: &BidegreeElement,
345        y: &BidegreeElement,
346    ) -> Option<BidegreeElement> {
347        let matrix = self.multiply_into(x, y.degree())?;
348        Some(combine_product(
349            &matrix,
350            y,
351            x.degree() + y.degree(),
352            self.prime(),
353        ))
354    }
355
356    /// The product `x · y`, where `x ∈ Ext(M, k)` and `y ∈ Ext(k, k)`. When `M == k` both operands
357    /// live in the same algebra $\Ext(k, k)$. The result lies in bidegree `x.degree() + y.degree()`.
358    ///
359    /// Panics if the product is out of the computed range; use
360    /// [`try_multiply`](Self::try_multiply) to handle that case.
361    pub fn multiply(&self, x: &BidegreeElement, y: &BidegreeElement) -> BidegreeElement {
362        self.try_multiply(x, y).expect(
363            "multiply: product is out of the computed range; compute further or use try_multiply",
364        )
365    }

368/// Build/cache the per-generator product map `res(source) → res(target)` for generator `g`.
369fn cached_generator_product_map<CC>(
370    products: &DashMap<BidegreeGenerator, Arc<ResolutionHomomorphism<CC, CC>>>,
371    source: &Arc<CC>,
372    target: &Arc<CC>,
373    g: BidegreeGenerator,
374) -> Arc<ResolutionHomomorphism<CC, CC>>
375where
376    CC: FreeChainComplex + AugmentedChainComplex,
377{
378    if let Some(map) = products.get(&g) {
379        return Arc::clone(&map);
380    }
381
382    let dim = source.number_of_gens_in_bidegree(g.degree());
383    let mut class = vec![0u32; dim];
384    class[g.idx()] = 1;
385
386    let name = format!("prod_{}_{}_{}", g.n(), g.s(), g.idx());
387    let hom = Arc::new(ResolutionHomomorphism::from_class(
388        name,
389        Arc::clone(source),
390        Arc::clone(target),
391        g.degree(),
392        &class,
393    ));
394
395    Arc::clone(products.entry(g).or_insert(hom).value())
396}

398/// The shared body of `multiply_into`: left-multiplication by `x` (a class in `Ext(source, k)`)
399/// applied to every generator of `Ext(target, k)` at bidegree `b`. Products land in `Ext(source,
400/// k)` at `b + x.degree()`. Returns `None` when out of the computed range.
401fn products_into<CC>(
402    source: &Arc<CC>,
403    target: &Arc<CC>,
404    products: &DashMap<BidegreeGenerator, Arc<ResolutionHomomorphism<CC, CC>>>,
405    prime: ValidPrime,
406    x: &BidegreeElement,
407    b: Bidegree,
408) -> Option<Matrix>
409where
410    CC: FreeChainComplex + AugmentedChainComplex,
411{
412    let shift = x.degree();
413    let result_deg = b + shift;
414
415    if !target.has_computed_bidegree(b) || !source.has_computed_bidegree(result_deg) {
416        return None;
417    }
418
419    let mult_dim = target.number_of_gens_in_bidegree(b);
420    let res_dim = source.number_of_gens_in_bidegree(result_deg);
421    let mut matrix = Matrix::new(prime, mult_dim, res_dim);
422
423    for (i, c) in x.vec().iter_nonzero() {
424        let map = cached_generator_product_map(
425            products,
426            source,
427            target,
428            BidegreeGenerator::new(shift, i),
429        );
430        map.extend_all();
431
432        // `hom_k(b.t())[j][k]`: `j` indexes the multiplicand generator of `Ext(k, k)` at `b`, `k`
433        // indexes the result generator of `Ext(source, k)` at `result_deg`.
434        let hom_k = map.get_map(result_deg.s()).hom_k(b.t());
435        for (j, row) in hom_k.iter().enumerate() {
436            for (k, &v) in row.iter().enumerate() {
437                matrix.row_mut(j).add_basis_element(k, c * v);
438            }
439        }
440    }
441    Some(matrix)
442}

444/// Combine the per-generator product `matrix` (rows indexed by generators of `y`'s bidegree) with
445/// the coordinates of `y` into the class `x · y` at bidegree `target`.
446fn combine_product(
447    matrix: &Matrix,
448    y: &BidegreeElement,
449    target: Bidegree,
450    prime: ValidPrime,
451) -> BidegreeElement {
452    let mut out = FpVector::new(prime, matrix.columns());
453    for (j, c) in y.vec().iter_nonzero() {
454        out.as_slice_mut().add(matrix.row(j), c);
455    }
456    BidegreeElement::new(target, out)
457}

482        let module = ExtModule::intrinsic(res);

563    /// Check that `class_product_map(x)` induces the same products as [`ExtModule::multiply_into`],

565    fn check_class_product_map<CC: FreeChainComplex + AugmentedChainComplex>(
566        module: &ExtModule<CC>,
567        x: &BidegreeElement,
568        max: Bidegree,
569    ) {
570        let algebra = module.algebra();
571        let map = algebra.class_product_map(x, max);
572        map.extend_all();
573
574        let mut compared = 0;
575        for b in algebra.resolution().iter_nonzero_stem() {
576            // `multiply_into` returns `None` once `b + x.degree()` is out of the computed range.
577            let Some(reference) = module.multiply_into(x, b) else {
578                continue;
579            };
580            let target = b + x.degree();
581            let hom_k = map.get_map(target.s()).hom_k(b.t());
582            assert_eq!(reference.rows(), hom_k.len());
583            for (j, row) in hom_k.iter().enumerate() {
584                let via_ref: Vec<u32> = reference.row(j).iter().collect();
585                assert_eq!(&via_ref, row, "product mismatch at multiplicand {b}");
586                compared += 1;
587            }
588        }
589        assert!(compared > 0, "expected at least one product comparison");
590    }

592    /// A multi-generator class takes the chain-level
593    /// [`ResolutionHomomorphism::linear_combination`] path; agreeing with
594    /// [`ExtModule::multiply_into`] pins `linear_combination`.
595    #[test]
596    fn test_class_product_map_multi_generator() {
597        let res = Arc::new(construct_standard::<false, _, _>("S_2", None).unwrap());
598        let max = Bidegree::n_s(20, 9);
599        res.compute_through_stem(max);
600        let module = ExtModule::intrinsic(res);
601
602        // (n = 15, s = 5) is the first bidegree of Ext(F_2, F_2) with two generators.
603        let x_deg = Bidegree::n_s(15, 5);
604        assert_eq!(module.algebra().dimension(x_deg), 2);
605        let x = module.algebra().element(x_deg, &[1, 1]);
606        check_class_product_map(&module, &x, max);
607    }
