1//! Enumeration of Milnor products at the prime 2, one anti-diagonal at a time.
2
3use super::PPart;
4
5/// The number of interior entries (not in row or column 0) on anti-diagonals of index at most
6/// [`PPart::MAX_LEN`].
7const MAX_CELLS: usize = PPart::MAX_LEN * (PPart::MAX_LEN - 1) / 2;
8
9/// Call `emit` with every p-part $T$ such that $P(T)$ appears in $P(r) P(s)$ at the prime 2, with
10/// multiplicity: equal terms may be emitted several times and should be summed mod 2.
11///
12/// $P(r) P(s)$ is the sum, over matrices $X$ with $\sum_j 2^j x_{ij} = r_i$ for each row and
13/// $\sum_i x_{ij} = s_j$ for each column, of $P(T(X))$, where $T_d$ is the sum of the $d$th
14/// anti-diagonal. Mod 2 a matrix contributes exactly when the entries of each anti-diagonal have
15/// pairwise disjoint binary digits.
16///
17/// We fill the interior entries one anti-diagonal at a time, from the last one down. The other
18/// entries of row $d$ and column $d$ lie on later anti-diagonals, so on reaching anti-diagonal $d$
19/// the slacks $x_{d0}$ and $x_{0d}$ are known. The anti-diagonal starts from their bitwise or, and
20/// each interior entry only takes values disjoint from the bits used so far, so every completed
21/// matrix contributes.
22///
23/// `degree` is an upper bound for the polynomial degree of the product. Entries on anti-diagonals
24/// $d$ with $2^d - 1 >$ `degree` must vanish, so we never visit them.
25pub(super) fn for_each_term(degree: i32, r: PPart, s: PPart, mut emit: impl FnMut(PPart)) {
26    let rows = r.len();
27    let cols = s.len();
28    if rows == 0 || cols == 0 {
29        emit(if rows == 0 { s } else { r });
30        return;
31    }
32    let mut max_diag = (rows + cols).min(PPart::MAX_LEN);
33    while max_diag > 1 && (1i64 << max_diag) - 1 > i64::from(degree) {
34        max_diag -= 1;
35    }
36
37    // What is left of `r` and `s`, indexed from 1. Once row (resp. column) `d` is filled, this is
38    // its entry in column (resp. row) 0.
39    let mut r_left = [0u32; PPart::MAX_LEN + 1];
40    let mut s_left = [0u32; PPart::MAX_LEN + 1];
41    for i in 0..rows {
42        r_left[i + 1] = r.get(i);
43    }
44    for j in 0..cols {
45        s_left[j + 1] = s.get(j);
46    }
47
48    // The interior cells in fill order, and whether each is the last of its anti-diagonal.
49    let mut row = [0u8; MAX_CELLS];
50    let mut col = [0u8; MAX_CELLS];
51    let mut last = [false; MAX_CELLS];
52    let mut num_cells = 0;
53    for d in (2..=max_diag).rev() {
54        for i in d.saturating_sub(cols).max(1)..=rows.min(d - 1) {
55            row[num_cells] = i as u8;
56            col[num_cells] = (d - i) as u8;
57            num_cells += 1;
58        }
59        last[num_cells - 1] = true;
60    }
61
62    let (a, b) = (r_left[max_diag], s_left[max_diag]);
63    if a & b != 0 {
64        return;
65    }
66    if max_diag == 1 {
67        emit(PPart::from_bits(u64::from(a | b) << PPart::shift(0)));
68        return;
69    }
70
71    // `value[k]` is the current entry of cell `k`, `bound[k]` the most it may be given the entries
72    // before it, and `before[k]` the bits used on its anti-diagonal by the cells before it.
73    let mut value = [0u32; MAX_CELLS];
74    let mut bound = [0u32; MAX_CELLS];
75    let mut before = [0u32; MAX_CELLS];
76    // The packed value of $T_2, \ldots$, filled in as anti-diagonals are completed.
77    let mut t_bits = 0u64;
78
79    let mut k = 0;
80    let mut tally = a | b;
81    'enter: loop {
82        // Value 0 never overlaps.
83        let (i, j) = (row[k] as usize, col[k] as usize);
84        before[k] = tally;
85        bound[k] = (r_left[i] >> j).min(s_left[j]);
86        value[k] = 0;
87        'advance: loop {
88            let used = before[k] | value[k];
89            if !last[k] {
90                k += 1;
91                tally = used;
92                continue 'enter;
93            }
94            // Anti-diagonal `d` is complete, and the slacks of `d - 1` are now known.
95            let d = (row[k] + col[k]) as usize;
96            let shift = PPart::shift(d - 1);
97            t_bits = (t_bits & !(u64::from(PPart::max_entry(d - 1)) << shift))
98                | (u64::from(used) << shift);
99            let (a, b) = (r_left[d - 1], s_left[d - 1]);
100            if a & b == 0 {
101                if d == 2 {
102                    emit(PPart::from_bits(
103                        t_bits | u64::from(a | b) << PPart::shift(0),
104                    ));
105                } else {
106                    k += 1;
107                    tally = a | b;
108                    continue 'enter;
109                }
110            }
111            // Backtrack to the latest cell that can still be increased.
112            loop {
113                let (i, j) = (row[k] as usize, col[k] as usize);
114                let v = value[k];
115                let mask = before[k];
116                let next = ((v | mask) + 1) & !mask;
117                if next <= bound[k] {
118                    r_left[i] -= (next - v) << j;
119                    s_left[j] -= next - v;
120                    value[k] = next;
121                    continue 'advance;
122                }
123                r_left[i] += v << j;
124                s_left[j] += v;
125                if k == 0 {
126                    return;
127                }
128                k -= 1;
129            }
130        }
131    }
132}

139    /// The polynomial degree of $P(x)$.
140    fn degree(x: PPart) -> i32 {
141        x.iter()
142            .enumerate()
143            .map(|(i, e)| ((1 << (i + 1)) - 1) * e as i32)
144            .sum()
145    }
146
147    /// The terms of `P(r) P(s)` that survive mod 2, sorted, by anti-diagonal enumeration and by
148    /// [`PPartMultiplier`].
149    fn both(r: PPart, s: PPart) -> (Vec<PPart>, Vec<PPart>) {
150        let degree = degree(r) + degree(s);
151        let reduce = |mut terms: Vec<PPart>| {
152            terms.sort_unstable();
153            let mut odd = Vec::new();
154            for chunk in terms.chunk_by(|a, b| a == b) {
155                if chunk.len() % 2 == 1 {
156                    odd.push(chunk[0]);
157                }
158            }
159            odd
160        };
161        let mut new = Vec::new();
162        for_each_term(degree, r, s, |t| new.push(t));
163        let mut multiplier = PPartMultiplier::<false>::new_from_allocation(
164            fp::prime::TWO,
165            r,
166            s,
167            PPartAllocation::default(),
168            0,
169            degree,
170        );
171        let mut old = Vec::new();
172        while let Some(c) = multiplier.next() {
173            if c % 2 == 1 {
174                old.push(multiplier.ans.p_part);
175            }
176        }
177        (reduce(new), reduce(old))
178    }

180    #[test]
181    fn small_products() {
182        let p = PPart::from_slice;
183        // Sq(1) Sq(1) = 0, Sq(1) Sq(2) = Sq(3), Sq(2) Sq(1) = Sq(3) + Sq(0, 1).
184        assert_eq!(both(p(&[1]), p(&[1])).0, vec![]);
185        assert_eq!(both(p(&[1]), p(&[2])).0, vec![p(&[3])]);
186        let (new, old) = both(p(&[2]), p(&[1]));
187        assert_eq!(new, old);
188        assert_eq!(new.len(), 2);
189        // Sq(4, 2) Sq(4, 1) = Sq(4, 2, 1) + Sq(7, 1, 1) + Sq(1, 3, 1).
190        let mut want = vec![p(&[4, 2, 1]), p(&[7, 1, 1]), p(&[1, 3, 1])];
191        want.sort_unstable();
192        assert_eq!(both(p(&[4, 2]), p(&[4, 1])).0, want);
193    }

195    /// Every pair of p-parts of length at most 4 with small entries.
196    #[test]
197    fn all_small_products_match() {
198        const BOUND: [u32; 4] = [12, 6, 3, 2];
199        let mut parts = vec![PPart::zero()];
200        for len in 1..=BOUND.len() {
201            let mut e = vec![0u32; len];
202            loop {
203                if e[len - 1] != 0 {
204                    parts.push(PPart::from_slice(&e));
205                }
206                let Some(i) = (0..len).find(|&i| e[i] < BOUND[i]) else {
207                    break;
208                };
209                e[..i].fill(0);
210                e[i] += 1;
211            }
212        }
213        let mut nonzero = 0;
214        for &r in &parts {
215            for &s in &parts {
216                let (new, old) = both(r, s);
217                assert_eq!(new, old, "P{r:?} P{s:?}");
218                nonzero += usize::from(!new.is_empty());
219            }
220        }
221        assert!(nonzero > 0);
222    }
