1//! The dual Steenrod algebras that a [`MilnorAlgebra`] can be a sub-Hopf-algebra of.
2
3use fp::prime::{TWO, ValidPrime};
4
5use super::{Exterior, MilnorAlgebra, MilnorAlgebraInner, MilnorProfile, NoExterior};
6use crate::algebra::Algebra;
7
8/// Which Hopf algebra a Steenrod algebra is a sub-Hopf-algebra of.
9///
10/// This is a property of the algebra rather than of its basis: the Adem basis only exists for the
11/// classical kind, and [`MilnorAlgebra::kind`] reports the kind of a Milnor one.
12///
13/// [`MilnorAlgebra::new`] builds the classical kind, and [`MilnorAlgebra::motivic_mod_tau`] and
14/// [`MilnorAlgebra::reduced_powers`] the others. Every pair of a prime and a
15/// [shape](super::MilnorShape) is exactly one kind, so the kind names what the shape encodes.
16#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
17pub enum SteenrodKind {
18    /// The Steenrod algebra $A$, at any prime.
19    Classical,
20    /// The mod-$\tau$ $\mathbb{C}$-motivic Steenrod algebra $A^{\mathbb{C}}/\tau$, at `p = 2`.
21    MotivicModTau,
22    /// The algebra of reduced powers $P = A/\!/E(Q_0, Q_1, \dots)$, at odd primes, whose dual is
23    /// $\mathbb{F}_p[\xi_1, \xi_2, \dots]$.
24    ReducedPowers,
25}
26
27impl SteenrodKind {
28    /// The kind with the given prime and shape.
29    pub(super) fn from_shape(p: ValidPrime, has_exterior: bool) -> Self {
30        match (p == 2, has_exterior) {
31            (true, false) | (false, true) => Self::Classical,
32            (true, true) => Self::MotivicModTau,
33            (false, false) => Self::ReducedPowers,
34        }
35    }
36}
37
38impl std::fmt::Display for SteenrodKind {
39    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
40        f.write_str(match self {
41            Self::Classical => "A",
42            Self::MotivicModTau => "A_C/tau",
43            Self::ReducedPowers => "P",
44        })
45    }
46}
47
48impl MilnorAlgebra {
49    /// $A^{\mathbb{C}}/\tau$ restricted to `profile`. The default profile gives all of it.
50    pub fn motivic_mod_tau(profile: MilnorProfile, unstable_enabled: bool) -> Self {
51        MilnorAlgebraInner::<Exterior>::new_with_profile(TWO, profile, unstable_enabled).into()
52    }
53
54    /// The algebra of reduced powers at `p` restricted to `profile`. The default profile gives all
55    /// of it.
56    ///
57    /// # Panics
58    ///
59    /// If `p = 2`, where the polynomial shape is the classical algebra.
60    pub fn reduced_powers(p: ValidPrime, profile: MilnorProfile, unstable_enabled: bool) -> Self {
61        assert!(p != 2, "the reduced powers are only defined at odd primes");
62        MilnorAlgebraInner::<NoExterior>::new_with_profile(p, profile, unstable_enabled).into()
63    }
64
65    /// The Hopf algebra this is a sub-Hopf-algebra of.
66    pub fn kind(&self) -> SteenrodKind {
67        SteenrodKind::from_shape(self.prime(), self.has_exterior())
68    }
69}
