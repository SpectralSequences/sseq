13    /// Return the degree and index of $Q_1^e P(x)$, or `None` if the element is not present
14    /// (e.g. out of range or excluded by the profile).
15    pub fn try_beps_pn(&self, e: u32, x: PPartEntry) -> Option<(i32, usize)> {
16        // `x` and `e` are unbounded, so compute the degree in `i64` and only narrow it once the
17        // bound has ruled the wide cases out.
18        let degree = self.q() as i64 * x as i64 + e as i64;
19        if degree > PPart::MAX_DEGREE as i64 || x > PPart::max_entry(0) {
20            return None;
21        }
22        let degree = degree as i32;
23        self.compute_basis(degree);
24        self.try_basis_element_to_index(&MilnorBasisElement {
25            degree,
26            q_part: e,
27            p_part: PPart::from_iter([x]),
28        })
29        .map(|index| (degree, index))
30    }
31
32    /// Return the degree and index of $Q_1^e P(x)$.
33    pub fn beps_pn(&self, e: u32, x: PPartEntry) -> (i32, usize) {
34        self.try_beps_pn(e, x).unwrap()
35    }

37    fn multiply_qpart(&self, m1: MilnorBasisElement, f: u32) -> Vec<(u32, MilnorBasisElement)> {
38        let mut new_result: Vec<(u32, MilnorBasisElement)> = vec![(1, m1)];
39        let mut old_result: Vec<(u32, MilnorBasisElement)> = Vec::new();
40
41        for k in BitflagIterator::set_bit_iterator(f as u64) {
42            let k = k as u32;
43            let pk = self.p.pow(k);
44            std::mem::swap(&mut new_result, &mut old_result);
45            new_result.clear();
46
47            // We implement the formula
48            // P(R) Q_k = Q_k P^R + Q_{k+1} P(R - p^k e_1) + Q_{k+2} P(R - p^k e_2) +
49            // ... + Q_{k + i} P(R - p^k e_i) + ...
50            // where e_i is the vector with value 1 in entry i and 0 otherwise (in the above
51            // formula, the first xi is xi_1, hence the offset below). If R - p^k e_i has a
52            // negative entry, the term is 0.
53            //
54            // We also use the fact that Q_k Q_j = -Q_j Q_k
55            for (coef, term) in &old_result {
56                for i in 0..=term.p_part.len() {
57                    // If there is already Q_{k+i} on the other side, the result is 0
58                    if term.q_part & (1 << (k + i as u32)) != 0 {
59                        continue;
60                    }
61                    let mut new_p = term.p_part;
62                    if i > 0 {
63                        // Check if R - p^k e_i < 0. Only do this from the first term onwards.
64                        let entry = new_p.get(i - 1);
65                        if entry < pk {
66                            continue;
67                        }
68                        new_p.set(i - 1, entry - pk);
69                    }
70
71                    // Now calculate the number of Q's we are moving past
72                    let larger_q = (term.q_part >> (k + i as u32 + 1)).count_ones();
73
74                    // Now put everything together
75                    let m = MilnorBasisElement {
76                        p_part: new_p,
77                        q_part: term.q_part | (1 << (k + i as u32)),
78                        degree: 0, // we don't really care about the degree here. The final degree of the whole calculation is known a priori
79                    };
80                    let c = if larger_q.is_multiple_of(2) {
81                        *coef
82                    } else {
83                        *coef * (self.prime() - 1)
84                    };
85
86                    new_result.push((c, m));
87                }
88            }
89        }
90        new_result
91    }

105    pub fn multiply_with_allocation(
106        &self,
107        mut res: FpSliceMut,
108        coef: u32,
109        m1: MilnorBasisElement,
110        m2: MilnorBasisElement,
111        excess: i32,
112        mut allocation: PPartAllocation,
113    ) -> PPartAllocation {
114        let target_deg = m1.degree + m2.degree;
115        if F::HAS_EXTERIOR {
116            let m1f = self.multiply_qpart(m1, m2.q_part);
117            for (cc, basis) in m1f {
118                allocation = self.multiply_ppart(
119                    res.copy(),
120                    cc * coef,
121                    basis.p_part,
122                    m2.p_part,
123                    basis.q_part,
124                    target_deg,
125                    excess,
126                    allocation,
127                );
128            }
129        } else {
130            allocation = self.multiply_ppart(
131                res, coef, m1.p_part, m2.p_part, 0, target_deg, excess, allocation,
132            );
133        }
134        allocation
135    }
136
137    /// Add `coef` times $P(r) P(s)$ to `res`, giving every term the exterior part `q_part`. The
138    /// result lies in degree `target_deg`, and terms of excess larger than `excess` are dropped.
139    ///
140    /// At the prime 2 the terms come from [`antidiagonal::for_each_term`], and otherwise from
141    /// [`PPartMultiplier`].
142    fn multiply_ppart(
143        &self,
144        mut res: FpSliceMut,
145        coef: u32,
146        r: PPart,
147        s: PPart,
148        q_part: u32,
149        target_deg: i32,
150        excess: i32,
151        allocation: PPartAllocation,
152    ) -> PPartAllocation {
153        let dim = self.dimension_unstable(target_deg, excess);
154        if self.prime() == 2 {
155            // The polynomial degree of the product is at most `target_deg / q`, with equality when
156            // there is no exterior part.
157            antidiagonal::for_each_term(target_deg / self.q(), r, s, |p_part| {
158                let idx = self.basis_element_to_index(&MilnorBasisElement {
159                    q_part,
160                    p_part,
161                    degree: target_deg,
162                });
163                if idx < dim {
164                    res.add_basis_element(idx, coef);
165                }
166            });
167            return allocation;
168        }
169
170        let mut multiplier = PPartMultiplier::<false>::new_from_allocation(
171            self.prime(),
172            r,
173            s,
174            allocation,
175            q_part,
176            target_deg,
177        );
178        while let Some(c) = multiplier.next() {
179            let idx = self.basis_element_to_index(&multiplier.ans);
180            if idx < dim {
181                res.add_basis_element(idx, c * coef);
182            }
183        }
184        multiplier.into_allocation()
185    }

187    pub fn multiply_basis_by_element(
188        &self,
189        res: FpSliceMut,
190        coef: u32,
191        m1: MilnorBasisElement,
192        s_deg: i32,
193        s: FpSlice,
194    ) {
195        PPartAllocation::with_local(|allocation| {
196            self.multiply_basis_by_element_with_allocation(res, coef, m1, s_deg, s, allocation)
197        });
198    }

200    pub(super) fn multiply_basis_by_element_with_allocation(
201        &self,
202        mut res: FpSliceMut,
203        coef: u32,
204        m1: MilnorBasisElement,
205        s_deg: i32,
206        s: FpSlice,
207        mut allocation: PPartAllocation,
208    ) -> PPartAllocation {
209        for (i, c) in s.iter_nonzero() {
210            allocation = self.multiply_with_allocation(
211                res.copy(),
212                coef * c,
213                m1,
214                self.basis_element_from_index(s_deg, i),
215                i32::MAX,
216                allocation,
217            );
218        }
219        allocation
220    }

304/// The least `l > k` whose binary digits are disjoint from those of `sum`, **given that `k` is
305/// itself disjoint from `sum`**.
306///
307/// Equivalently, the least such `l` with $\binom{\mathrm{sum} + l}{l}$ odd: adding `l` to `sum`
308/// carries exactly where they share a set bit, and the 2-adic valuation of that binomial is the
309/// number of carries (Kummer). So this steps straight to the next value that keeps a Milnor
310/// coefficient non-zero mod 2, instead of testing candidates and discarding them.
311///
312/// # Panics
313///
314/// In debug builds, if `k & sum != 0`. The identity genuinely needs it: the increment is allowed
315/// to carry through `k`'s own bits but not through `sum`'s, so a `k` that overlaps `sum` can come
316/// back *smaller* than `k` (`next_disjoint(2, 2) == 1`). Every caller is walking a matrix whose
317/// entries are already pairwise disjoint along each anti-diagonal, so the precondition holds.
318///
319/// The result can exceed any bound the caller has in mind; compare it against that separately.
320pub const fn next_disjoint(sum: PPartEntry, k: PPartEntry) -> PPartEntry {
321    debug_assert!(k & sum == 0, "next_disjoint needs k disjoint from sum");
322    ((k | sum) + 1) & !sum
323}

345    #[allow(unused_mut)] // Mut is only used with odd primes
346    pub fn new_from_allocation(
347        p: ValidPrime,
348        r: PPart,
349        s: PPart,
350        mut allocation: PPartAllocation,
351        q_part: u32,
352        degree: i32,
353    ) -> Self {
354        if MOD4 {
355            assert_eq!(p, 2);
356        }
357        let rows = r.len() + 1;
358        let cols = s.len() + 1;
359        let diag_num = r.len() + s.len();
360        #[cfg(feature = "odd-primes")]
361        {
362            allocation.diagonal.clear();
363            allocation.diagonal.reserve_exact(std::cmp::max(rows, cols));
364        }
365
366        let mut M = allocation.m;
367        M.reset(rows, cols);
368
369        for i in 1..rows {
370            M[i][0] = r.entry(i - 1);
371        }
372        // Iterated rather than indexed because the loop variable would index the single row
373        // `M[0]`, unlike the loop above, which indexes a different row each time.
374        for (k, entry) in M[0][1..cols].iter_mut().enumerate() {
375            *entry = s.entry(k);
376        }
377
378        let ans = MilnorBasisElement {
379            q_part,
380            p_part: PPart::zero(),
381            degree,
382        };
383        Self {
384            #[cfg(feature = "odd-primes")]
385            diagonal: allocation.diagonal,
386            p,
387            M,
388            r,
389            rows,
390            cols,
391            diag_num,
392            ans,
393            init: true,
394        }
395    }

398        PPartAllocation {

400            #[cfg(feature = "odd-primes")]
401            diagonal: self.diagonal,
402        }

405    /// This compute the first l > k such that (sum + l) choose l != 0 mod p, stopping if we reach
406    /// max + 1. This is useful for incrementing the matrix.
407    ///
408    /// TODO: Improve odd prime performance
409    fn next_val(&self, sum: PPartEntry, k: PPartEntry, max: PPartEntry) -> PPartEntry {
410        match self.prime().as_u32() {
411            2 => {
412                if MOD4 {
413                    // x.count_ones() + y.count_ones() - (x + y).count_ones() is the number of
414                    // carries when adding x to y.
415                    //
416                    // The p-adic valuation of (n + r) choose r is the number of carries when
417                    // adding r to n in base p.
418                    (k + 1..max + 1)
419                        .find(|&l| {
420                            sum & l == 0
421                                || (sum.count_ones() + l.count_ones()) - (sum + l).count_ones() == 1
422                        })
423                        .unwrap_or(max + 1)
424                } else {
425                    next_disjoint(sum, k)
426                }
427            }
428            _ => (k + 1..max + 1)
429                .find(|&l| !PPartEntry::binomial_odd_is_zero(self.prime(), sum + l, l))
430                .unwrap_or(max + 1),
431        }
432    }
433
434    /// We have a matrix of the form
435    ///    | s₁  s₂  s₃ ...
436    /// --------------------
437    /// r₁ |
438    /// r₂ |     x_{ij}
439    /// r₃ |
440    ///
441    /// We think of ourselves as modifiying the center pieces x_{ij}, while the r_i's and s_j's are
442    /// only there to ensure the x_{ij}'s don't get too big. The idea is to sweep through the
443    /// matrix row by row, from top-to-bottom, and left-to-right. In each pass, we find the first
444    /// entry that can be incremented. We then increment it and zero out all the entries that
445    /// appear before it. This will give us all valid entries.
446    fn update(&mut self) -> bool {
447        for i in 1..self.rows {
448            // total is sum x_{ij} p^j up to the jth column
449            let mut total = self.M[i][0];
450            let mut p_to_the_j = 1;
451            for j in 1..self.cols {
452                p_to_the_j *= self.prime().as_u32();
453                if total < p_to_the_j {
454                    // We don't have enough weight left in the entries above this one in the column to increment this cell.
455                    // Add the weight from this cell to the total, we can use it to increment a cell lower down.
456                    total += self.M[i][j] * p_to_the_j;
457                    continue;
458                }
459                let col_sum: PPartEntry = (0..i).map(|k| self.M[k][j]).sum();
460                if col_sum == 0 {
461                    total += self.M[i][j] * p_to_the_j;
462                    continue;
463                }
464
465                let max_inc = std::cmp::min(col_sum, total / p_to_the_j);
466
467                // Compute the sum of entries along the diagonal to the bottom-left
468                let mut sum = 0;
469                for c in (i + j + 1).saturating_sub(self.rows)..j {
470                    sum += self.M[i + j - c][c];
471                }
472
473                // Find the next possible value we can increment M[i][j] to without setting the
474                // coefficient to 0. The coefficient is the multinomial coefficient of the
475                // diagonal, and if the multinomial coefficient of any subset is zero, so is the
476                // coefficient of the whole diagonal.
477                let next_val = self.next_val(sum, self.M[i][j], max_inc + self.M[i][j]);
478                let inc = next_val - self.M[i][j];
479
480                // The remaining obstacle to incrementing this entry is the column sum condition.
481                // For this, we only need a non-zero entry in the column j above row i.
482                if inc <= max_inc {
483                    // If so, we found our next matrix.
484                    for row in 1..i {
485                        self.M[row][0] = self.r.entry(row - 1);
486                        for col in 1..self.cols {
487                            self.M[0][col] += self.M[row][col];
488                            self.M[row][col] = 0;
489                        }
490                    }
491                    for col in 1..j {
492                        self.M[0][col] += self.M[i][col];
493                        self.M[i][col] = 0;
494                    }
495                    self.M[0][j] -= inc;
496                    self.M[i][j] += inc;
497                    self.M[i][0] = total - p_to_the_j * inc;
498                    return true;
499                }
500                // All the cells above this one are zero so we didn't find our next matrix.
501                // Add the weight from this cell to the total, we can use it to increment a cell lower down.
502                total += self.M[i][j] * p_to_the_j;
503            }
504        }
505        false
506    }
507}

509impl<const MOD4: bool> Iterator for PPartMultiplier<MOD4> {
510    type Item = u32;
511
512    fn next(&mut self) -> Option<u32> {
513        let p = self.prime().as_u32();
514        'outer: loop {
515            let mut coef = 1;
516
517            if self.init {
518                self.init = false;
519                for i in 1..std::cmp::min(self.cols, self.rows) {
520                    if MOD4 {
521                        coef *= PPartEntry::binomial4(self.M[i][0] + self.M[0][i], self.M[0][i]);
522                        coef %= 4;
523                    } else {
524                        coef *= PPartEntry::binomial(
525                            self.prime(),
526                            self.M[i][0] + self.M[0][i],
527                            self.M[0][i],
528                        );
529                        coef %= p;
530                    }
531                    if coef == 0 {
532                        continue 'outer;
533                    }
534                }
535                // The answer is the top row of the matrix plus `r`, entrywise. Accumulate into
536                // a plain word and store once.
537                let mut ans = 0;
538                for i in 0..std::cmp::max(self.cols, self.rows) - 1 {
539                    let mut entry = self.r.entry(i);
540                    if i + 1 < self.cols {
541                        entry += self.M[0][i + 1];
542                    }
543                    debug_assert!(entry <= PPart::max_entry(i));
544                    ans |= (entry as u64) << PPart::shift(i);
545                }
546                self.ans.p_part = PPart::from_bits(ans);
547                return Some(coef);
548            } else if self.update() {
549                let mut ans = 0;
550                for diag_idx in 1..=self.diag_num {
551                    let i_min = (diag_idx + 1).saturating_sub(self.cols);
552                    let i_max = std::cmp::min(diag_idx + 1, self.rows);
553                    let mut sum = 0;
554
555                    if self.prime() == 2 {
556                        if MOD4 {
557                            for i in i_min..i_max {
558                                let entry = self.M[i][diag_idx - i];
559                                sum += entry;
560                                if coef.is_multiple_of(2) {
561                                    coef *= PPartEntry::binomial2(sum, entry);
562                                } else {
563                                    coef *= PPartEntry::binomial4(sum, entry);
564                                }
565                                coef %= 4;
566                                if coef == 0 {
567                                    continue 'outer;
568                                }
569                            }
570                        } else {
571                            let mut or = 0;
572                            for i in i_min..i_max {
573                                sum += self.M[i][diag_idx - i];
574                                or |= self.M[i][diag_idx - i];
575                            }
576                            if sum != or {
577                                continue 'outer;
578                            }
579                        }
580                    } else {
581                        #[cfg(feature = "odd-primes")]
582                        {
583                            self.diagonal.clear();
584                            for i in i_min..i_max {
585                                self.diagonal.push(self.M[i][diag_idx - i]);
586                                sum += self.M[i][diag_idx - i];
587                            }
588
589                            coef *= PPartEntry::multinomial_odd(self.prime(), &mut self.diagonal);
590                            coef %= p;
591                            if coef == 0 {
592                                continue 'outer;
593                            }
594                        }
595                    }
596                    // `diag_num` counts diagonals of the working matrix, which can exceed the
597                    // number of entries a p-part of this degree can have; those trailing
598                    // diagonals are necessarily zero and need not be stored.
599                    if diag_idx <= PPart::MAX_LEN {
600                        debug_assert!(sum <= PPart::max_entry(diag_idx - 1));
601                        ans |= (sum as u64) << PPart::shift(diag_idx - 1);
602                    } else {
603                        debug_assert_eq!(sum, 0);
604                    }
605                }
606                self.ans.p_part = PPart::from_bits(ans);
607
608                return Some(coef);
609            } else {
610                return None;
611            }
612        }
613    }

624    /// The product computed by walking every matrix with [`PPartMultiplier`], at every prime.
625    fn reference_multiply<F: MilnorShape>(
626        algebra: &MilnorAlgebraInner<F>,
627        mut res: FpSliceMut,
628        m1: MilnorBasisElement,
629        m2: MilnorBasisElement,
630        excess: i32,
631    ) {
632        let target_deg = m1.degree + m2.degree;
633        let factors = if F::HAS_EXTERIOR {
634            algebra.multiply_qpart(m1, m2.q_part)
635        } else {
636            vec![(1, m1)]
637        };
638        for (cc, basis) in factors {
639            let mut multiplier = PPartMultiplier::<false>::new_from_allocation(
640                algebra.prime(),
641                basis.p_part,
642                m2.p_part,
643                PPartAllocation::default(),
644                basis.q_part,
645                target_deg,
646            );
647            while let Some(c) = multiplier.next() {
648                let idx = algebra.basis_element_to_index(&multiplier.ans);
649                if idx < algebra.dimension_unstable(target_deg, excess) {
650                    res.add_basis_element(idx, c * cc);
651                }
652            }
653        }
654    }

656    /// Check every product of basis elements of total degree at most `max_degree` against
657    /// [`reference_multiply`], stably and, if the algebra supports it, at every excess. Returns the
658    /// number of non-zero products, so that callers can check the comparison was not vacuous.
659    fn check_against_reference<F: MilnorShape>(
660        algebra: &MilnorAlgebraInner<F>,
661        max_degree: i32,
662    ) -> usize {
663        algebra.compute_basis(max_degree);
664        let mut nonzero = 0;
665        for r_deg in 0..=max_degree {
666            for s_deg in 0..=max_degree - r_deg {
667                let out_deg = r_deg + s_deg;
668                let excesses: Vec<i32> = if algebra.unstable_enabled {
669                    (0..=out_deg).collect()
670                } else {
671                    vec![i32::MAX]
672                };
673                for r_idx in 0..algebra.dimension(r_deg) {
674                    for s_idx in 0..algebra.dimension(s_deg) {
675                        let m1 = algebra.basis_element_from_index(r_deg, r_idx);
676                        let m2 = algebra.basis_element_from_index(s_deg, s_idx);
677                        for &excess in &excesses {
678                            let mut want =
679                                FpVector::new(algebra.prime(), algebra.dimension(out_deg));
680                            let mut got = want.clone();
681                            reference_multiply(algebra, want.as_slice_mut(), m1, m2, excess);
682                            if excess == i32::MAX {
683                                algebra.multiply_basis_elements(
684                                    got.as_slice_mut(),
685                                    1,
686                                    r_deg,
687                                    r_idx,
688                                    s_deg,
689                                    s_idx,
690                                );
691                            } else {
692                                algebra.multiply_basis_elements_unstable(
693                                    got.as_slice_mut(),
694                                    1,
695                                    r_deg,
696                                    r_idx,
697                                    s_deg,
698                                    s_idx,
699                                    excess,
700                                );
701                            }
702                            assert_eq!(got, want, "{m1} * {m2} at excess {excess}");
703                            nonzero += usize::from(!want.is_zero());
704                        }
705                    }
706                }
707            }
708        }
709        nonzero
710    }

741        let p = ValidPrime::new(2);
742
743        // On the full algebra, `try_beps_pn` agrees with the panicking `beps_pn`

745        let algebra = MilnorAlgebra::new(p, false);
746        for x in 1..16 {

749        }

751        // On A(2) (profile [3, 2, 1], truncated), the first xi exponent is bounded
752        // by 2^3 - 1 = 7, so P(7) is present but P(8) is excluded by the profile.
753        let a2 = MilnorAlgebra::new_with_profile(
754            p,
755            MilnorProfile {
756                q_part: !0,
757                p_part: vec![3, 2, 1],
758                truncated: true,
759            },
760            false,
761        );
762        // Valid input: agrees with `beps_pn`.
763        assert_eq!(a2.try_beps_pn(0, 7), Some(a2.beps_pn(0, 7)));
764        // Invalid input: excluded by the profile, so `None` instead of a panic.
765        assert_eq!(a2.try_beps_pn(0, 8), None);

780    /// `next_disjoint` is the jump-to-valid form of the "is this binomial odd" test that the
781    /// Milnor coefficient needs; check it against the brute-force search it replaces, over every
782    /// input satisfying its precondition.
783    #[test]
784    fn next_disjoint_matches_brute_force() {
785        for sum in 0..64u32 {
786            for k in (0..64u32).filter(|k| k & sum == 0) {
787                let expected = (k + 1..)
788                    .find(|l| l & sum == 0)
789                    .expect("a disjoint value always exists");
790                assert_eq!(next_disjoint(sum, k), expected, "sum = {sum}, k = {k}");
791                // The characterisation it is actually used for.
792                assert_ne!(u32::binomial2(sum + expected, expected), 0);
793            }
794        }
795    }
