12impl MilnorAlgebra {
13    /// The sub-Hopf-algebra of this algebra's kind with the given profile, at the same prime, or
14    /// `None` if the profile is not valid. Unstable support is off.
15    ///
16    /// This need not lie in `self`; see [`Self::is_subalgebra_of`].
17    pub fn with_profile(&self, profile: MilnorProfile) -> Option<Self> {
18        if !profile.is_valid() {
19            return None;
20        }
21        let p = self.prime();
22        Some(match self {
23            Self::Polynomial(_) => {
24                MilnorAlgebraInner::<NoExterior>::new_with_profile(p, profile, false).into()
25            }
26            Self::Exterior(_) => {
27                MilnorAlgebraInner::<Exterior>::new_with_profile(p, profile, false).into()
28            }
29        })
30    }

32    /// The trivial sub-Hopf-algebra of this algebra's kind, the ground field.
33    pub fn ground_field(&self) -> Self {
34        let profile = MilnorProfile {
35            truncated: true,
36            q_part: 0,
37            p_part: vec![],
38        };
39        self.with_profile(profile).unwrap()
40    }

42    /// Whether `self` is a sub-Hopf-algebra of `other`: the same kind and prime, and a smaller
43    /// profile.
44    pub fn is_subalgebra_of(&self, other: &Self) -> bool {
45        let len = std::cmp::max(self.profile().p_part.len(), other.profile().p_part.len());
46        self.kind() == other.kind()
47            && self.prime() == other.prime()
48            && self.exterior_part() & !other.exterior_part() == 0
49            // One entry past the longer profile compares the two tails.
50            && (0..=len).all(|i| self.profile().get_p_part(i) <= other.profile().get_p_part(i))
51    }
52
53    /// The $Q_k$ in this algebra, as a mask of the q-part. Only the exterior shape has any.
54    fn exterior_part(&self) -> u32 {
55        if self.has_exterior() {
56            self.profile().q_part
57        } else {
58            0
59        }
60    }

62    /// `p^{e_i}`, which bounds entry `i` of the p-part of every basis element, or `None` if that
63    /// entry is unbounded. A bound past [`PPartEntry::MAX`] bounds nothing.
64    fn modulus(&self, i: usize) -> Option<PPartEntry> {
65        self.prime()
66            .as_u32()
67            .checked_pow(self.profile().get_p_part(i))
68    }

70    /// The degree of the top basis element, or `None` if the algebra is infinite or that degree
71    /// does not fit in an `i32`.
72    ///
73    /// A finite sub-Hopf-algebra is a Poincaré duality algebra, so the top basis element is the
74    /// only one in its degree.
75    pub fn top_degree(&self) -> Option<i32> {
76        let p = self.prime();
77        let profile = self.profile();
78        if !profile.truncated || profile.p_part.len() > combinatorics::MAX_XI {
79            return None;
80        }
81        let tau_degrees = combinatorics::tau_degrees(p);
82        let xi_degrees = combinatorics::xi_degrees(p);
83
84        let mut top = 0i64;
85        for k in BitflagIterator::set_bit_iterator(self.exterior_part() as u64) {
86            // A saturated entry stands for a degree past `i32::MAX`.
87            if tau_degrees[k] == i32::MAX {
88                return None;
89            }
90            top += tau_degrees[k] as i64;
91        }
92        for (i, &d) in xi_degrees.iter().enumerate().take(profile.p_part.len()) {
93            top += self.q() as i64 * (self.modulus(i)? as i64 - 1) * d as i64;
94        }
95        i32::try_from(top).ok()
96    }
97
98    /// The dimension of the algebra, saturating at `u64::MAX`, or `None` where
99    /// [`Self::top_degree`] is.
100    pub fn total_dimension(&self) -> Option<u64> {
101        self.top_degree()?;
102        let exterior = 1u64 << self.exterior_part().count_ones();
103        Some(
104            (0..self.profile().p_part.len())
105                .map(|i| self.modulus(i).unwrap() as u64)
106                .fold(exterior, u64::saturating_mul),
107        )
108    }

110    /// The slope of the vanishing line of Ext over this algebra, or `None` where
111    /// [`Self::top_degree`] is: $\mathrm{Ext}^{s, t}$ vanishes for
112    /// `t >= slope * (s + 1) + top_degree` ([Nassau](https://arxiv.org/abs/1910.04063), Theorem
113    /// 3.1).
114    ///
115    /// This is the slope of the steepest polynomial generator of the May spectral sequence: `v_k`
116    /// for each `Q_k`, and for each `ξ_i^{p^j}` either `h_{i, j}` at `p = 2` or `b_{i, j}` at odd
117    /// primes. The exterior `h_{i, j}` at odd primes are bounded by the top degree.
118    pub fn vanishing_slope(&self) -> Option<i64> {
119        self.top_degree()?;
120        let p = self.prime().as_i32() as i64;
121        let q = self.q() as i64;
122        let tau_degrees = combinatorics::tau_degrees(self.prime());
123        let xi_degrees = combinatorics::xi_degrees(self.prime());
124
125        let polynomial = self
126            .profile()
127            .p_part
128            .iter()
129            .enumerate()
130            .flat_map(|(i, &e)| {
131                (0..e).map(move |j| {
132                    let h = q * xi_degrees[i] as i64 * p.pow(j);
133                    if p == 2 { h } else { h * p / 2 }
134                })
135            });
136        BitflagIterator::set_bit_iterator(self.exterior_part() as u64)
137            .map(|k| tau_degrees[k] as i64)
138            .chain(polynomial)
139            .max()
140            .or(Some(0))
141    }

143    /// The component in this algebra of `elt`, a basis element of an algebra containing it.
144    ///
145    /// The component of $Q(E)P(R)$ is $Q(E \cap E')P(R \bmod p^e)$, where $E'$ and $e$ are this
146    /// algebra's q-part and profile, and an unbounded entry of $e$ leaves that entry of $R$ alone.
147    /// Grouping the basis of the larger algebra by component splits it into translates of the
148    /// elements whose component is $1$, one per basis element of this algebra, which is the
149    /// freeness of the larger algebra over this one.
150    pub fn component(&self, elt: &MilnorBasisElement) -> MilnorBasisElement {
151        let mut component = MilnorBasisElement {
152            q_part: elt.q_part & self.exterior_part(),
153            p_part: PPart::zero(),
154            degree: 0,
155        };
156        for i in 0..PPart::MAX_LEN {
157            let entry = elt.p_part.get(i);
158            component
159                .p_part
160                .set(i, self.modulus(i).map_or(entry, |m| entry % m));
161        }
162        self.compute_degree(&mut component);
163        component
164    }
165
166    /// Whether the component of `elt` is `component`, without building it.
167    pub fn has_component(&self, elt: &MilnorBasisElement, component: &MilnorBasisElement) -> bool {
168        elt.q_part & self.exterior_part() == component.q_part
169            && (0..PPart::MAX_LEN).all(|i| {
170                let entry = elt.p_part.get(i);
171                self.modulus(i).map_or(entry, |m| entry % m) == component.p_part.get(i)
172            })
173    }
174
175    /// A mask that compiles [`Self::has_component`] to `elt.p_part.bits() & mask ==
176    /// component.p_part.bits()`. It exists when the basis has no exterior part and every modulus
177    /// is a power of two, i.e. at the polynomial shape at `p = 2`.
178    ///
179    /// Each entry occupies a fixed field of the packed word, so its residue mod $2^e$ is the low
180    /// $e$ bits of that field.
181    pub fn packed_component_mask(&self) -> Option<u64> {
182        if self.kind() != SteenrodKind::Classical || self.prime() != 2 {
183            return None;
184        }
185        Some((0..PPart::MAX_LEN).fold(0, |mask, i| {
186            // A profile entry wider than the field, unbounded ones included, constrains all of it.
187            let width = std::cmp::min(self.profile().get_p_part(i), PPart::width(i));
188            mask | ((1u64 << width) - 1) << PPart::shift(i)
189        }))
190    }
191}

193/// The whole algebra as its kind and prime, e.g. `A(p=2)`, and otherwise `F_p`, `A(n)` or
194/// `E(Q_0..Q_n)` where the profile is one of those, and the profile itself where not. An unbounded
195/// tail is written `..`.
196impl<F: MilnorShape> std::fmt::Display for MilnorAlgebraInner<F> {
197    fn fmt(&self, out: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
198        let p = self.prime();
199        let profile = self.profile();
200        if *profile == MilnorProfile::default() {
201            return write!(
202                out,
203                "{}(p={p})",
204                SteenrodKind::from_shape(p, F::HAS_EXTERIOR)
205            );
206        }
207        let MilnorProfile {
208            truncated,
209            q_part,
210            p_part,
211        } = profile;
212        let len = p_part.len();
213        let is_an = *truncated && p_part.iter().rev().copied().eq(1..=len as PPartEntry);
214        let entries = p_part
215            .iter()
216            .map(|&e| {
217                if e == PPartEntry::MAX {
218                    "inf".to_string()
219                } else {
220                    e.to_string()
221                }
222            })
223            .chain((!truncated).then(|| "..".to_string()))
224            .join(",");
225        if F::HAS_EXTERIOR {
226            let top_q = (u32::BITS - q_part.leading_zeros()) as usize;
227            let q_is_initial = q_part.count_ones() as usize == top_q;
228            if *truncated && *q_part == 0 && len == 0 {
229                write!(out, "F_{p}")
230            } else if is_an && q_is_initial && top_q == len + 1 {
231                write!(out, "A({len})")
232            } else if *truncated && len == 0 && q_is_initial {
233                write!(out, "E(Q_0..Q_{})", top_q - 1)
234            } else {
235                write!(out, "B({q_part:#b}; {entries})")
236            }
237        } else if *truncated && len == 0 {
238            write!(out, "F_{p}")
239        } else if is_an {
240            write!(out, "A({})", len - 1)
241        } else {
242            write!(out, "B({entries})")
243        }
244    }
245}

255    /// The sub-Hopf-algebra of `ambient`'s kind with the given finite profile.
256    fn sub(ambient: &MilnorAlgebra, q_part: u32, p_part: &[u32]) -> MilnorAlgebra {
257        let profile = MilnorProfile {
258            truncated: true,
259            q_part,
260            p_part: p_part.to_vec(),
261        };
262        ambient.with_profile(profile).unwrap()
263    }

265    /// Check `ambient`'s components in `sub` through `max_degree`, and if `sub` is finite its top
266    /// degree and dimension against its basis.
267    fn check(ambient: &MilnorAlgebra, sub: &MilnorAlgebra, max_degree: i32) {
268        assert!(sub.is_subalgebra_of(ambient));
269
270        let top = sub.top_degree();
271        if let Some(top) = top {
272            sub.compute_basis(top + 4);
273            assert_eq!(sub.dimension(top), 1, "top degree of {:?}", sub.profile());
274            assert!((top + 1..=top + 4).all(|t| sub.dimension(t) == 0));
275            let total: usize = (0..=top).map(|t| sub.dimension(t)).sum();
276            assert_eq!(sub.total_dimension(), Some(total as u64));
277        }
278
279        // `counts[t][c]` is the number of basis elements of `ambient` in degree `t` whose
280        // component is `c`, given by its `(degree, index)` in `sub`.
281        ambient.compute_basis(max_degree);
282        sub.compute_basis(max_degree);
283        let mut counts = vec![HashMap::new(); max_degree as usize + 1];
284        for t in 0..=max_degree {
285            for idx in 0..ambient.dimension(t) {
286                let elt = ambient.basis_element_from_index(t, idx);
287                let component = sub.component(&elt);
288                let index = sub
289                    .try_basis_element_to_index(&component)
290                    .unwrap_or_else(|| panic!("component {component} of {elt} is not in sub"));
291                assert!(sub.has_component(&elt, &component));
292                if let Some(mask) = sub.packed_component_mask() {
293                    assert_eq!(elt.p_part.bits() & mask, component.p_part.bits());
294                }
295                *counts[t as usize]
296                    .entry((component.degree, index))
297                    .or_insert(0usize) += 1;
298            }
299        }
300
301        // Freeness: the elements with component `c` are the translates of those with component 1.
302        let top = top.unwrap_or(max_degree);
303        for t in 0..=max_degree {
304            for k in 0..=std::cmp::min(t, top) {
305                for index in 0..sub.dimension(k) {
306                    let translates = counts[(t - k) as usize].get(&(0, 0)).copied();
307                    let actual = counts[t as usize].get(&(k, index)).copied();
308                    assert_eq!(
309                        actual.unwrap_or(0),
310                        translates.unwrap_or(0),
311                        "degree {t}, component ({k}, {index})"
312                    );
313                }
314            }
315        }
316    }

323    /// Profiles of the classical algebra at `p = 2`, in it and in each other.
324    #[test]
325    fn test_polynomial_at_two() {
326        let ambient = ambient_at_two();
327        for p_part in [&[1][..], &[2, 1], &[2, 2, 1], &[3, 2, 1]] {
328            check(&ambient, &sub(&ambient, 0, p_part), 40);
329        }
330        // Nested finite algebras, A(1) in A(2).
331        let a2 = sub(&ambient, 0, &[3, 2, 1]);
332        check(&a2, &sub(&ambient, 0, &[2, 1]), 23);
333
334        assert_eq!(sub(&ambient, 0, &[2, 1]).top_degree(), Some(6));
335        assert_eq!(a2.top_degree(), Some(23));
336        assert_eq!(a2.total_dimension(), Some(64));
337    }

370    /// $A^{\mathbb{C}}/\tau$ and the odd primary algebras, where the q-part counts.
371    #[cfg(feature = "odd-primes")]
372    #[test]
373    fn test_exterior() {
374        let c_tau = MilnorAlgebra::motivic_mod_tau(MilnorProfile::default(), false);
375        for (q_part, p_part) in [(0b1, &[][..]), (0b11, &[]), (0b11, &[1]), (0b111, &[2, 1])] {
376            check(&c_tau, &sub(&c_tau, q_part, p_part), 30);
377        }
378        assert_eq!(sub(&c_tau, 0b11, &[1]).top_degree(), Some(6));
379        assert_eq!(c_tau.top_degree(), None);
380        // A(1) of A_C/tau is not a subalgebra of the classical A(2).
381        assert!(!sub(&c_tau, 0b11, &[1]).is_subalgebra_of(&ambient_at_two()));
382
383        let p3 = MilnorAlgebra::new(fp::prime::ValidPrime::new(3), false);
384        assert!(!sub(&ambient_at_two(), 0, &[2, 1]).is_subalgebra_of(&p3));
385        for (q_part, p_part) in [(0b1, &[][..]), (0b11, &[1])] {
386            check(&p3, &sub(&p3, q_part, p_part), 60);
387        }
388        assert_eq!(sub(&p3, 0b11, &[1]).top_degree(), Some(14));
389
390        // |Q_19| = 2 * 3^19 - 1 does not fit in an `i32`, even on its own.
391        assert_eq!(sub(&p3, 1 << 19, &[]).top_degree(), None);
392
393        // Containment compares both parts.
394        let a1 = sub(&p3, 0b11, &[1]);
395        assert!(sub(&p3, 0b11, &[]).is_subalgebra_of(&a1));
396        assert!(!a1.is_subalgebra_of(&sub(&p3, 0b11, &[])));
397        assert!(!sub(&p3, 0b111, &[]).is_subalgebra_of(&a1));
398
399        // The reduced powers are their own kind, not a subalgebra of A.
400        let reduced = MilnorAlgebra::reduced_powers(p3.prime(), MilnorProfile::default(), false);
401        assert_eq!(reduced.kind(), SteenrodKind::ReducedPowers);
402        assert!(!reduced.is_subalgebra_of(&p3));
403        check(&reduced, &sub(&reduced, 0, &[2, 1]), 60);
404    }
