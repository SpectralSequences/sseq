3/// The exponent sequence $(r_1, r_2, \ldots)$ of a Milnor basis element $P(r_1, r_2, \ldots)$,
4/// bit-packed into a single `u64`.
5///
6/// Entry $r_{i+1}$ occupies `WIDTHS[i]` bits starting at bit `SHIFTS[i]` (both private). The
7/// widths are forced by the degree bound: at $p = 2$ the internal degree of $P(R)$ is
8/// $\sum_i r_i (2^i - 1)$ and every term is non-negative, so an element of degree at most
9/// [`Self::MAX_DEGREE`] has $r_i \le \mathrm{MAX\\_DEGREE}/(2^i - 1)$. At an odd prime the same
10/// argument bounds $r_i$ by that quantity divided by $q = 2(p-1)$, so the $p = 2$ widths are valid
11/// for every prime and this type is prime-agnostic.
12///
13/// Trailing zeros are not represented: $P(2, 1)$ and $P(2, 1, 0)$ have the same packed value. That
14/// is what makes the packed value a canonical key, and it makes [`Self::len`] the position of the
15/// highest non-zero entry rather than a stored field.
16///
17/// # Invariant
18///
19/// Every entry fits in its field. This holds for any element of degree at most
20/// [`Self::MAX_DEGREE`], which [`MilnorAlgebra`](super::MilnorAlgebra)'s
21/// [`compute_basis`](crate::algebra::Algebra::compute_basis) enforces up front, so the packing can
22/// never silently truncate. [`Self::set`] asserts it anyway, and [`Self::try_from_slice`]
23/// reports failure instead of panicking for input that has not been through that gate.
24///
25/// The derived ordering compares the packed words. That is a total order and consistent with
26/// equality, which is all a sorted table needs, but it is *not* lexicographic in the exponents:
27/// entry `i` sits at [`Self::shift`]`(i)`, so `r_1` is the least significant field and therefore
28/// the last tie-breaker. Callers that need the exponents ordered lexicographically must compare
29/// [`Self::iter`] instead.
30#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Default)]
31pub struct PPart(u64);
32
33impl PPart {
34    /// `FIELD_OF_BIT[b]` is the index of the entry owning bit `b`, letting [`Self::len`] turn a
35    /// `leading_zeros` into an entry index without looping.
36    const FIELD_OF_BIT: [u8; 64] = {
37        let mut table = [0; 64];
38        let mut i = 0;
39        while i < Self::MAX_LEN {
40            let mut b = Self::SHIFTS[i];
41            while b < Self::SHIFTS[i + 1] {
42                table[b as usize] = i as u8;
43                b += 1;
44            }
45            i += 1;
46        }
47        table
48    };
49    /// The largest internal degree whose exponent sequences are guaranteed to fit.
50    ///
51    /// This is the largest bound for which the field widths sum to at most 64. It is far beyond
52    /// anything currently reachable.
53    pub const MAX_DEGREE: i32 = 2045;
54    /// The number of entries that can be stored. This equals `fp`'s `MAX_MULTINOMIAL_LEN`.
55    pub const MAX_LEN: usize = 10;
56    /// `SHIFTS[i]` is the bit offset of entry `i`; `SHIFTS[MAX_LEN]` is the total width, 64.
57    const SHIFTS: [u32; Self::TABLE_LEN] = {
58        let mut shifts = [0; Self::TABLE_LEN];
59        let mut i = 0;
60        while i < Self::MAX_LEN {
61            shifts[i + 1] = shifts[i] + Self::WIDTHS[i];
62            i += 1;
63        }
64        shifts
65    };
66    /// The length of the layout tables.
67    ///
68    /// This is [`Self::MAX_LEN`] rounded up to a power of two so that [`Self::entry`] can mask its
69    /// index instead of bounds-checking it; entries at or past `MAX_LEN` are given width 0, so they
70    /// read as zero.
71    const TABLE_LEN: usize = 16;
72    /// `WIDTHS[i]` is the number of bits holding $r_{i+1}$: the number of bits needed to represent
73    /// `MAX_DEGREE / (2^(i+1) - 1)`.
74    const WIDTHS: [u32; Self::TABLE_LEN] = [11, 10, 9, 8, 7, 6, 5, 4, 3, 1, 0, 0, 0, 0, 0, 0];
75
76    /// The largest value entry `i` can hold.
77    pub const fn max_entry(i: usize) -> PPartEntry {
78        ((1u64 << Self::WIDTHS[i]) - 1) as PPartEntry
79    }
80
81    /// The number of bits holding entry `i`.
82    ///
83    /// Together with [`Self::shift`] this lets callers build a mask over [`Self::bits`] directly,
84    /// e.g. to test many entries in one comparison.
85    pub const fn width(i: usize) -> u32 {
86        Self::WIDTHS[i]
87    }
88
89    /// The bit offset of entry `i` within [`Self::bits`].
90    pub const fn shift(i: usize) -> u32 {
91        Self::SHIFTS[i]
92    }
93
94    const fn mask(i: usize) -> u64 {
95        ((1u64 << Self::WIDTHS[i]) - 1) << Self::SHIFTS[i]
96    }
97
98    pub const fn zero() -> Self {
99        Self(0)
100    }
101
102    /// The raw packed value.
103    ///
104    /// Two exponent sequences are equal exactly when their bits are, so this is a complete hash
105    /// key, and it can be compared against a packed mask in one operation (see
106    /// `MilnorSubalgebra::packed_signature` in `ext`).
107    pub const fn bits(self) -> u64 {
108        self.0
109    }
110
111    /// Reinterpret a raw packed value.
112    ///
113    /// Callers that assemble entries by shifting must uphold the type invariant themselves: each
114    /// entry must lie within its field, which holds for any element of degree at most
115    /// [`Self::MAX_DEGREE`]. This exists so hot loops can accumulate into a plain `u64` and store
116    /// once, rather than read-modify-write through [`Self::set`] per entry.
117    pub(crate) const fn from_bits(bits: u64) -> Self {
118        Self(bits)
119    }
120
121    /// Entry `i`, for `i < TABLE_LEN`, with no bounds check.
122    #[inline]
123    pub(super) const fn entry(self, i: usize) -> PPartEntry {
124        debug_assert!(i < Self::TABLE_LEN);
125        let i = i & (Self::TABLE_LEN - 1);
126        ((self.0 >> Self::SHIFTS[i]) & ((1 << Self::WIDTHS[i]) - 1)) as PPartEntry
127    }
128
129    /// Entry `i`, or 0 if `i` is past the end. Accepts any index.
130    #[inline]
131    pub const fn get(self, i: usize) -> PPartEntry {
132        if i >= Self::MAX_LEN { 0 } else { self.entry(i) }
133    }
134
135    /// Set entry `i` to `v`.
136    ///
137    /// # Panics
138    ///
139    /// If `i >= MAX_LEN`, or `v` does not fit in entry `i`.
140    #[inline]
141    pub fn set(&mut self, i: usize, v: PPartEntry) {
142        assert!(i < Self::MAX_LEN, "p-part index {i} out of range");
143        assert!(
144            v <= Self::max_entry(i),
145            "p-part entry {v} does not fit in the {} bits at index {i}",
146            Self::WIDTHS[i],
147        );
148        self.0 = (self.0 & !Self::mask(i)) | ((v as u64) << Self::SHIFTS[i]);
149    }
150
151    /// The number of entries up to and including the last non-zero one.
152    #[inline]
153    pub const fn len(self) -> usize {
154        if let Some(idx) = self.0.highest_one() {
155            Self::FIELD_OF_BIT[idx as usize] as usize + 1
156        } else {
157            0
158        }
159    }
160
161    #[inline]
162    pub const fn is_empty(self) -> bool {
163        self.0 == 0
164    }
165
166    /// Zero every entry from `n` onwards, i.e. the packed form of `self[..n]`.
167    #[inline]
168    pub const fn truncate(self, n: usize) -> Self {
169        if n >= Self::MAX_LEN {
170            self
171        } else {
172            Self(self.0 & ((1 << Self::SHIFTS[n]) - 1))
173        }
174    }
175
176    /// Iterate over the entries in order from lowest to highest index.
177    pub fn iter(self) -> impl DoubleEndedIterator<Item = PPartEntry> + ExactSizeIterator {
178        (0..self.len()).map(move |i| self.get(i))
179    }
180
181    /// Pack `entries`, returning `None` if they do not fit. Use this for anything derived from
182    /// user input; use [`Self::from_slice`] when the degree bound already guarantees a fit.
183    pub fn try_from_slice(entries: &[PPartEntry]) -> Option<Self> {
184        let mut result = Self::zero();
185        for (i, &entry) in entries.iter().enumerate() {
186            // A zero past the end is just padding, which the packed form drops anyway.
187            if entry == 0 {
188                continue;
189            }
190            if i >= Self::MAX_LEN || entry > Self::max_entry(i) {
191                return None;
192            }
193            result.set(i, entry);
194        }
195        Some(result)
196    }
197
198    /// Pack `entries`, panicking if they do not fit.
199    pub fn from_slice(entries: &[PPartEntry]) -> Self {
200        Self::try_from_slice(entries).unwrap_or_else(|| {
201            panic!(
202                "p-part {entries:?} exceeds the degree {} bound",
203                Self::MAX_DEGREE
204            )
205        })
206    }
207}

209impl FromIterator<PPartEntry> for PPart {
210    fn from_iter<I: IntoIterator<Item = PPartEntry>>(iter: I) -> Self {
211        let mut result = Self::zero();
212        for (i, entry) in iter.into_iter().enumerate() {
213            if entry != 0 {
214                result.set(i, entry);
215            }
216        }
217        result
218    }
219}

223        f.debug_list().entries(self.iter()).finish()

232    /// The packing is only sound because each field is wide enough for every entry that can occur
233    /// at degree at most `MAX_DEGREE`. Check that against the $\xi$-degrees directly, so that
234    /// changing `MAX_DEGREE` or `WIDTHS` without the other fails loudly.
235    #[test]
236    fn ppart_widths_cover_max_degree() {
237        let xi_degrees = combinatorics::xi_degrees(fp::prime::TWO);
238        for (i, &xi_degree) in xi_degrees.iter().enumerate().take(PPart::MAX_LEN) {
239            // deg P(R) = sum_i r_i (2^i - 1) with non-negative terms, so r_i <= deg / (2^i - 1).
240            let bound = PPart::MAX_DEGREE / xi_degree;
241            assert!(
242                bound <= PPart::max_entry(i) as i32,
243                "entry {i} needs to hold {bound} but only holds up to {}",
244                PPart::max_entry(i),
245            );
246        }
247        // There is no entry beyond `MAX_LEN` to store: the xi-degree table itself stops there, so
248        // `compute_ppart` cannot produce a longer p-part. If `fp` ever raises
249        // `MAX_MULTINOMIAL_LEN`, this fires and `WIDTHS` has to be revisited.
250        assert_eq!(xi_degrees.len(), PPart::MAX_LEN);
251        // ... and the layout uses the whole word, so `MAX_DEGREE` is as large as it can be.
252        assert_eq!(
253            PPart::shift(PPart::MAX_LEN - 1) + PPart::width(PPart::MAX_LEN - 1),
254            64
255        );
256    }
257
258    #[test]
259    fn ppart_accessors() {
260        let mut p = PPart::from_slice(&[3, 0, 5]);
261        assert_eq!(p.len(), 3);
262        assert_eq!(p.iter().collect::<Vec<_>>(), vec![3, 0, 5]);
263        assert_eq!(p.get(1), 0);
264        assert_eq!(p.get(2), 5);
265        // Reading past the end is zero, not a panic.
266        assert_eq!(p.get(7), 0);
267        assert_eq!(p.get(PPart::MAX_LEN), 0);
268
269        // Trailing zeros are not represented, so they do not affect equality, length or hashing.
270        assert_eq!(PPart::from_slice(&[3, 0, 5, 0, 0]), p);
271        assert_eq!(PPart::from_slice(&[]), PPart::zero());
272        assert_eq!(PPart::from_slice(&[0, 0]), PPart::zero());
273        assert_eq!(PPart::zero().len(), 0);
274        assert!(PPart::zero().is_empty());
275
276        assert_eq!(p.truncate(2), PPart::from_slice(&[3]));
277        assert_eq!(p.truncate(0), PPart::zero());
278        assert_eq!(p.truncate(PPart::MAX_LEN + 3), p);
279
280        p.set(1, 7);
281        assert_eq!(p, PPart::from_slice(&[3, 7, 5]));
282        p.set(2, 0);
283        assert_eq!(p, PPart::from_slice(&[3, 7]));
284    }

286    #[test]
287    fn ppart_rejects_out_of_range() {
288        // Too many entries, and an entry too large for its field.
289        assert_eq!(PPart::try_from_slice(&[1; PPart::MAX_LEN + 1]), None);
290        assert_eq!(PPart::try_from_slice(&[0, PPart::max_entry(1) + 1]), None);
291        // ... but a zero past the end is only padding.
292        let mut padded = vec![0; PPart::MAX_LEN + 4];
293        padded[0] = 2;
294        assert_eq!(
295            PPart::try_from_slice(&padded),
296            Some(PPart::from_slice(&[2]))
297        );
298    }
