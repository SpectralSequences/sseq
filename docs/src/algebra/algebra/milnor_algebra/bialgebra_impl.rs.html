4impl MilnorAlgebra {
5    /// Advance `element` to the next p-part bounded entrywise by `max`, in odometer order.
6    ///
7    /// Returns `true` once the odometer wraps, i.e. when `element` was already `max`.
8    ///
9    /// This carries *before* incrementing rather than after. The two orders enumerate the same
10    /// sequence, but incrementing first would transiently store `max[i] + 1`, which need not fit
11    /// in a packed field whose width is exactly saturated by `max[i]`.
12    fn increment_p_part(element: &mut PPart, max: PPart) -> bool {
13        for i in 0..max.len() {
14            let entry = element.get(i);
15            if entry < max.get(i) {
16                element.set(i, entry + 1);
17                return false;
18            }
19            element.set(i, 0);
20        }
21        true
22    }
23}

25impl Bialgebra for MilnorAlgebra {
26    fn coproduct(&self, op_deg: i32, op_idx: usize) -> Vec<(i32, usize, i32, usize)> {
27        assert_eq!(self.prime(), 2, "Coproduct at odd primes not supported");
28        if op_deg == 0 {
29            return vec![(0, 0, 0, 0)];
30        }
31        let xi_degrees = combinatorics::xi_degrees(self.prime());
32
33        let p_part = self.basis_element_from_index(op_deg, op_idx).p_part;
34
35        let mut len: usize = 1;
36        for i in p_part.iter() {
37            len *= i as usize + 1;
38        }
39        let mut result = Vec::with_capacity(len);
40
41        let n = p_part.len();
42        let mut cur_ppart = PPart::zero();
43        loop {
44            let mut left_degree: i32 = 0;
45            let mut right_ppart = PPart::zero();
46            for (i, &xi_degree) in xi_degrees.iter().enumerate().take(n) {
47                let entry = cur_ppart.get(i);
48                left_degree += entry as i32 * xi_degree;
49                right_ppart.set(i, p_part.get(i) - entry);
50            }
51            let right_degree: i32 = op_deg - left_degree;
52
53            let left_idx = self.basis_element_to_index(&MilnorBasisElement {
54                degree: left_degree,
55                q_part: 0,
56                p_part: cur_ppart,
57            });
58            let right_idx = self.basis_element_to_index(&MilnorBasisElement {
59                degree: right_degree,
60                q_part: 0,
61                p_part: right_ppart,
62            });
63
64            result.push((left_degree, left_idx, right_degree, right_idx));
65            if Self::increment_p_part(&mut cur_ppart, p_part) {
66                break;
67            }
68        }
69        result
70    }
71
72    fn decompose(&self, op_deg: i32, op_idx: usize) -> Vec<(i32, usize)> {
73        vec![(op_deg, op_idx)]
74    }
75}

77#[cfg(test)]
78mod tests {
79    use super::*;

81    /// `increment_p_part` walks up to and including `max`, whose top entry may saturate its field.
82    /// Incrementing before carrying would overflow there.
83    #[test]
84    fn ppart_odometer_handles_saturated_field() {
85        let top = PPart::MAX_LEN - 1;
86        let mut max = PPart::from_slice(&[2]);
87        max.set(top, PPart::max_entry(top));
88
89        let mut count = 0;
90        let mut cur = PPart::zero();
91        loop {
92            count += 1;
93            if MilnorAlgebra::increment_p_part(&mut cur, max) {
94                break;
95            }
96        }
97        assert_eq!(count, 3 * (PPart::max_entry(top) as usize + 1));
98        // Wrapping leaves the odometer back at zero.
99        assert_eq!(cur, PPart::zero());
100    }
