8/// Flat, contiguous storage for the "seqno" (hash-free index) computation.
9///
10/// Row-major with a fixed `width` (the number of ξ-degrees), so entry `(e, h)` lives at
11/// `g[e * width + h]`; degrees `0..=max_degree` are populated.
12pub(super) struct SeqnoTables {
13    max_degree: i32,
14    width: usize,
15    g: Vec<usize>,
16}
17
18/// A borrowed view of the seqno tables, acquired once for a batch of lookups.
19///
20/// Holding this pins the revision of the tables that was current at acquisition, so
21/// the per-lookup cost is the rank itself with no atomic. It is therefore valid only for
22/// the degrees that revision covered: a ranker held across a concurrent grow will not see
23/// the new degrees, and ranking one panics.
24pub struct SeqnoRanker {
25    tables: Arc<SeqnoTables>,
26    xi: &'static [i32],
27}
28
29impl SeqnoRanker {
30    /// The index of `P(p_part)` in the Milnor basis of `degree`.
31    #[inline]
32    pub fn rank(&self, p_part: PPart, degree: i32) -> usize {
33        let t = &*self.tables;
34        let w = t.width;
35        debug_assert_eq!(
36            degree,
37            p_part
38                .iter()
39                .zip(self.xi)
40                .map(|(r, &x)| r as i32 * x)
41                .sum::<i32>(),
42            "degree {degree} does not match the p-part {p_part:?}"
43        );
44        debug_assert!(
45            degree <= t.max_degree,
46            "degree {degree} exceeds seqno tables built to {}",
47            t.max_degree
48        );
49        let mut cur_d = degree;
50        let mut rank = 0;
51        for h in (1..p_part.len()).rev() {
52            let r = p_part.get(h) as i32;
53            if r == 0 {
54                continue;
55            }
56            let below = cur_d - r * self.xi[h];
57            rank += t.g[cur_d as usize * w + h] - t.g[below as usize * w + h];
58            cur_d = below;
59        }
60        rank
61    }
62}
63
64impl<F: MilnorShape> MilnorAlgebraInner<F> {
65    /// Whether seqno (hash-free index) can be used.
66    pub(super) fn seqno_applicable(&self) -> bool {
67        !F::HAS_EXTERIOR
68            && self.p == fp::prime::TWO
69            && !self.unstable_enabled
70            && self.profile.is_trivial()
71    }
72
73    /// Index of `elt`, or `None` if it is not a basis element of the computed range.
74    ///
75    /// [`Self::seqno`] trusts its input, so this rejects an element with a Q part, a degree that
76    /// does not match its p-part, or a degree beyond the tables.
77    pub(super) fn try_seqno(&self, elt: &MilnorBasisElement) -> Option<usize> {
78        let xi = combinatorics::xi_degrees(self.p);
79        let degree: i32 = elt.p_part.iter().zip(xi).map(|(r, &x)| r as i32 * x).sum();
80        let in_range = self
81            .seqno_tables
82            .load()
83            .as_ref()
84            .is_some_and(|t| (0..=t.max_degree).contains(&elt.degree));
85        (elt.q_part == 0 && degree == elt.degree && in_range)
86            .then(|| self.seqno(elt.p_part, elt.degree))
87    }
88
89    /// Build the flat SeqnoTables up to `max_degree`.
90    ///
91    /// Idempotent: if the stored tables already reach `max_degree` this returns immediately;
92    /// otherwise it rebuilds the whole table and atomically swaps it in.
93    pub fn compute_seqno_tables(&self, max_degree: i32) {
94        assert!(self.seqno_applicable());
95        assert!(
96            (0..=PPart::MAX_DEGREE).contains(&max_degree),
97            "seqno tables only supported for degrees 0..={}, got {max_degree}",
98            PPart::MAX_DEGREE,
99        );
100        if let Some(t) = &*self.seqno_tables.load()
101            && t.max_degree >= max_degree
102        {
103            return;
104        }
105
106        let xi = combinatorics::xi_degrees(self.prime());
107        let width = xi.len();
108        let rows = max_degree as usize + 1;
109
110        let mut n = vec![0usize; rows * width];
111        for e in 0..=max_degree {
112            let base = e as usize * width;
113            for m in 0..width {
114                let without = if m == 0 {
115                    (e == 0) as usize
116                } else {
117                    n[base + m - 1]
118                };
119                let with = if xi[m] <= e {
120                    n[(e - xi[m]) as usize * width + m]
121                } else {
122                    0
123                };
124                n[base + m] = without + with;
125            }
126        }
127
128        let mut g = vec![0usize; rows * width];
129        for e in 0..=max_degree {
130            let base = e as usize * width;
131            for h in 1..width {
132                let head = n[base + h - 1];
133                let tail = if xi[h] <= e {
134                    g[(e - xi[h]) as usize * width + h]
135                } else {
136                    0
137                };
138                g[base + h] = head + tail;
139            }
140        }
141
142        let new_tables = Arc::new(SeqnoTables {
143            max_degree,
144            width,
145            g,
146        });
147        self.seqno_tables.rcu(|current| match current.as_deref() {
148            Some(t) if t.max_degree >= max_degree => current.clone(),
149            _ => Some(new_tables.clone()),
150        });
151    }
152
153    /// A handle to acquire seqno tables once for a batch of lookups.
154    pub fn seqno_ranker(&self) -> SeqnoRanker {
155        debug_assert!(self.seqno_applicable());
156        SeqnoRanker {
157            tables: self
158                .seqno_tables
159                .load_full()
160                .expect("seqno tables not built; call compute_seqno_tables first"),
161            xi: combinatorics::xi_degrees(self.prime()),
162        }
163    }
164
165    /// The index of `P(p_part)` in the Milnor basis of `degree`.
166    ///
167    /// Computed hash-free from precomputed tables. Assumes seqno is applicable and that
168    /// `p_part` is a genuine basis element (trimmed, in range) of `degree`.
169    pub fn seqno(&self, p_part: PPart, degree: i32) -> usize {
170        self.seqno_ranker().rank(p_part, degree)
171    }
172}
