8/// Flat, contiguous storage for the "seqno" (hash-free index) computation.
9///
10/// Row-major with a fixed `width` (the number of ξ-degrees), so entry `(e, h)` lives at
11/// `g[e * width + h]`; degrees `0..=max_degree` are populated.
12pub struct SeqnoTables {
13    max_degree: i32,
14    width: usize,
15    g: Vec<usize>,
16    /// The degrees of the $\xi_i$ the tables were built from.
17    xi: &'static [i32],
18}
19
20impl SeqnoTables {
21    /// The index of `P(p_part)` in the Milnor basis of `degree`.
22    ///
23    /// `p_part` must be a basis element of `degree`, and `degree` at most the degree the tables
24    /// were built to.
25    #[inline]
26    pub fn rank(&self, p_part: PPart, degree: i32) -> usize {
27        let w = self.width;
28        debug_assert_eq!(
29            degree,
30            p_part
31                .iter()
32                .zip(self.xi)
33                .map(|(r, &x)| r as i32 * x)
34                .sum::<i32>(),
35            "degree {degree} does not match the p-part {p_part:?}"
36        );
37        debug_assert!(
38            degree <= self.max_degree,
39            "degree {degree} exceeds seqno tables built to {}",
40            self.max_degree
41        );
42        let mut cur_d = degree;
43        let mut rank = 0;
44        for h in (1..p_part.len()).rev() {
45            let r = p_part.get(h) as i32;
46            if r == 0 {
47                continue;
48            }
49            let below = cur_d - r * self.xi[h];
50            rank += self.g[cur_d as usize * w + h] - self.g[below as usize * w + h];
51            cur_d = below;
52        }
53        rank
54    }
55}
56
57impl<F: MilnorShape> MilnorAlgebraInner<F> {
58    /// Whether seqno (hash-free index) can be used.
59    pub(super) fn seqno_applicable(&self) -> bool {
60        !F::HAS_EXTERIOR
61            && self.p == fp::prime::TWO
62            && !self.unstable_enabled
63            && self.profile.is_trivial()
64    }
65
66    /// Index of `elt`, or `None` if it is not a basis element of the computed range.
67    ///
68    /// [`Self::seqno`] trusts its input, so this rejects an element with a Q part, a degree that
69    /// does not match its p-part, or a degree beyond the tables.
70    pub(super) fn try_seqno(&self, elt: &MilnorBasisElement) -> Option<usize> {
71        let guard = self.seqno_tables.load();
72        let t = guard.as_deref()?;
73        let degree: i32 = elt
74            .p_part
75            .iter()
76            .zip(t.xi)
77            .map(|(r, &x)| r as i32 * x)
78            .sum();
79        (elt.q_part == 0 && degree == elt.degree && (0..=t.max_degree).contains(&elt.degree))
80            .then(|| t.rank(elt.p_part, elt.degree))
81    }
82
83    /// Call `f` with the seqno tables, holding a single guard for all of its lookups.
84    ///
85    /// # Panics
86    ///
87    /// If the tables have not been built.
88    pub(super) fn with_seqno_tables<R>(&self, f: impl FnOnce(&SeqnoTables) -> R) -> R {
89        let guard = self.seqno_tables.load();
90        f(guard
91            .as_deref()
92            .expect("seqno tables not built; call compute_seqno_tables first"))
93    }
94
95    /// Build the flat SeqnoTables up to `max_degree`.
96    ///
97    /// Idempotent: if the stored tables already reach `max_degree` this returns immediately;
98    /// otherwise it rebuilds the whole table and atomically swaps it in.
99    pub fn compute_seqno_tables(&self, max_degree: i32) {
100        assert!(self.seqno_applicable());
101        assert!(
102            (0..=PPart::MAX_DEGREE).contains(&max_degree),
103            "seqno tables only supported for degrees 0..={}, got {max_degree}",
104            PPart::MAX_DEGREE,
105        );
106        if let Some(t) = &*self.seqno_tables.load()
107            && t.max_degree >= max_degree
108        {
109            return;
110        }
111
112        let xi = combinatorics::xi_degrees(self.prime());
113        let width = xi.len();
114        let rows = max_degree as usize + 1;
115
116        let mut n = vec![0usize; rows * width];
117        for e in 0..=max_degree {
118            let base = e as usize * width;
119            for m in 0..width {
120                let without = if m == 0 {
121                    (e == 0) as usize
122                } else {
123                    n[base + m - 1]
124                };
125                let with = if xi[m] <= e {
126                    n[(e - xi[m]) as usize * width + m]
127                } else {
128                    0
129                };
130                n[base + m] = without + with;
131            }
132        }
133
134        let mut g = vec![0usize; rows * width];
135        for e in 0..=max_degree {
136            let base = e as usize * width;
137            for h in 1..width {
138                let head = n[base + h - 1];
139                let tail = if xi[h] <= e {
140                    g[(e - xi[h]) as usize * width + h]
141                } else {
142                    0
143                };
144                g[base + h] = head + tail;
145            }
146        }
147
148        let new_tables = Arc::new(SeqnoTables {
149            max_degree,
150            width,
151            g,
152            xi,
153        });
154        self.seqno_tables.rcu(|current| match current.as_deref() {
155            Some(t) if t.max_degree >= max_degree => current.clone(),
156            _ => Some(new_tables.clone()),
157        });
158    }
159
160    /// The current seqno tables, to rank a batch of elements with [`SeqnoTables::rank`].
161    ///
162    /// Cloning the `Arc` touches a reference count shared by all threads, so acquire the tables
163    /// once per batch rather than once per lookup. They cover only the degrees computed when they
164    /// were acquired: tables held across a concurrent grow will not see the new degrees.
165    ///
166    /// # Panics
167    ///
168    /// If the tables have not been built.
169    pub fn seqno_tables(&self) -> Arc<SeqnoTables> {
170        debug_assert!(self.seqno_applicable());
171        self.seqno_tables
172            .load_full()
173            .expect("seqno tables not built; call compute_seqno_tables first")
174    }
175
176    /// The index of `P(p_part)` in the Milnor basis of `degree`.
177    ///
178    /// Computed hash-free from precomputed tables. Assumes seqno is applicable and that
179    /// `p_part` is a genuine basis element (trimmed, in range) of `degree`.
180    pub fn seqno(&self, p_part: PPart, degree: i32) -> usize {
181        self.with_seqno_tables(|t| t.rank(p_part, degree))
182    }
183}
