37impl<const U: bool, M: Module> MuFreeModuleHomomorphism<U, M>
38where
39    M::Algebra: MuAlgebra<U>,
40{
41    /// Apply to one basis element, taking its `(operation, generator)` pair from `cursor`.
42    ///
43    /// The degree is the cursor's, so a caller cannot pair a cursor with a mismatched degree. There
44    /// is no length check on `result`; the callers check it, and differently for the restricted
45    /// variants.
46    fn apply_to_basis_element_with(
47        &self,
48        cursor: &mut OpGenCursor<'_, U, M::Algebra>,
49        result: FpSliceMut,
50        coeff: u32,
51        input_index: usize,
52    ) {
53        let OperationGeneratorPair {
54            operation_degree,
55            generator_degree,
56            operation_index,
57            generator_index,
58        } = cursor.get(input_index);
59
60        if generator_degree >= self.min_degree() {
61            let output_on_generator = self.output(generator_degree, generator_index);
62            self.target.act(
63                result,
64                coeff,
65                operation_degree,
66                operation_index,
67                generator_degree - self.degree_shift,
68                output_on_generator.as_slice(),
69            );
70        }
71    }
72}

74impl<const U: bool, M: Module> ModuleHomomorphism for MuFreeModuleHomomorphism<U, M>
75where
76    M::Algebra: MuAlgebra<U>,
77{
78    type Source = MuFreeModule<U, M::Algebra>;
79    type Target = M;
80
81    fn source(&self) -> Arc<Self::Source> {
82        Arc::clone(&self.source)
83    }
84
85    fn target(&self) -> Arc<Self::Target> {
86        Arc::clone(&self.target)
87    }
88
89    fn degree_shift(&self) -> i32 {
90        self.degree_shift
91    }
92
93    /// Builds a cursor for this one element. Prefer [`Self::apply`] or [`Self::get_matrix`] when
94    /// applying to several elements of a degree, so that one cursor serves them all.
95    fn apply_to_basis_element(
96        &self,
97        result: FpSliceMut,
98        coeff: u32,
99        input_degree: i32,
100        input_index: usize,
101    ) {
102        assert_eq!(
103            self.target.dimension(input_degree - self.degree_shift),
104            result.as_slice().len()
105        );
106        self.apply_to_basis_element_restricted(result, coeff, input_degree, input_index)
107    }
108
109    /// Reuses one cursor across the terms of `input`, which `iter_nonzero` yields in ascending
110    /// order.
111    fn apply(&self, mut result: FpSliceMut, coeff: u32, input_degree: i32, input: FpSlice) {
112        assert!(input_degree >= self.source.min_degree());
113        assert_eq!(
114            self.target.dimension(input_degree - self.degree_shift),
115            result.as_slice().len()
116        );
117        let p = self.prime();
118        let mut cursor = self.source.opgen_cursor(input_degree);
119        for (i, v) in input.iter_nonzero() {
120            self.apply_to_basis_element_with(&mut cursor, result.copy(), (coeff * v) % p, i);
121        }
122    }
123
124    /// Builds one cursor per worker, which then walks that worker's rows in ascending order.
125    fn get_matrix(&self, mut matrix: MatrixSliceMut, degree: i32) {
126        assert_eq!(self.source.dimension(degree), matrix.rows());
127        assert_eq!(
128            self.target.dimension(degree - self.degree_shift),
129            matrix.columns()
130        );
131
132        if matrix.columns() == 0 {
133            return;
134        }
135
136        let source = &*self.source;
137        matrix.maybe_par_iter_mut().enumerate().for_each_init(
138            || source.opgen_cursor(degree),
139            |cursor, (i, row)| self.apply_to_basis_element_with(cursor, row, 1, i),
140        );
141    }
142
143    /// See [`Self::get_matrix`] for the cursor.
144    ///
145    /// The columns are the target's dimension in the SHIFTED degree, matching what
146    /// [`Self::get_matrix`] asserts and what each row is then filled to.
147    fn get_partial_matrix(&self, degree: i32, inputs: &[usize]) -> Matrix {
148        self.get_partial_matrix_restricted(
149            degree,
150            inputs,
151            self.target.dimension(degree - self.degree_shift),
152        )
153    }
154
155    fn quasi_inverse(&self, degree: i32) -> Option<&QuasiInverse> {
156        self.quasi_inverses.get(degree).and_then(Option::as_ref)
157    }
158
159    fn kernel(&self, degree: i32) -> Option<&Subspace> {
160        self.kernels.get(degree).and_then(Option::as_ref)
161    }
162
163    fn image(&self, degree: i32) -> Option<&Subspace> {
164        self.images.get(degree).and_then(Option::as_ref)
165    }
166
167    fn compute_auxiliary_data_through_degree(&self, degree: i32) {
168        self.kernels.extend(degree, |i| {
169            let (image, kernel, qi) = self.auxiliary_data(i);
170            self.images.push_checked(Some(image), i);
171            self.quasi_inverses.push_checked(Some(qi), i);
172            Some(kernel)
173        });
174    }
175}

181    pub fn new(
182        source: Arc<MuFreeModule<U, M::Algebra>>,
183        target: Arc<M>,
184        degree_shift: i32,
185    ) -> Self {
186        let min_degree = std::cmp::max(source.min_degree(), target.min_degree() + degree_shift);
187        let outputs = OnceBiVec::new(min_degree);
188        let kernels = OnceBiVec::new(min_degree);
189        let images = OnceBiVec::new(min_degree);
190        let quasi_inverses = OnceBiVec::new(min_degree);
191        Self {
192            source,
193            target,
194            outputs,
195            images,
196            kernels,
197            quasi_inverses,
198            min_degree,
199            degree_shift,
200        }
201    }
202
203    pub fn degree_shift(&self) -> i32 {
204        self.degree_shift
205    }
206
207    pub fn min_degree(&self) -> i32 {
208        self.min_degree
209    }
210
211    pub fn next_degree(&self) -> i32 {
212        self.outputs.len()
213    }
214
215    pub fn output(&self, generator_degree: i32, generator_index: usize) -> &FpVector {
216        assert!(
217            generator_degree >= self.min_degree(),
218            "generator_degree {} less than min degree {}",
219            generator_degree,
220            self.min_degree()
221        );
222        assert!(
223            generator_index < self.source.number_of_gens_in_degree(generator_degree),
224            "generator_index {} greater than number of generators {}",
225            generator_index,
226            self.source.number_of_gens_in_degree(generator_degree)
227        );
228        &self.outputs[generator_degree][generator_index]
229    }
230
231    /// A truncating variant of [`ModuleHomomorphism::apply_to_basis_element`] that allows `result`
232    /// to span only a prefix of the target's degree-`(input_degree - degree_shift)` basis, rather
233    /// than the whole thing.
234    ///
235    /// The caller must guarantee that the image of the basis element is supported within that
236    /// prefix; otherwise `act` will panic on an out-of-bounds write. The Nassau resolution in the
237    /// `ext` crate relies on this.
238    pub fn apply_to_basis_element_restricted(
239        &self,
240        result: FpSliceMut,
241        coeff: u32,
242        input_degree: i32,
243        input_index: usize,
244    ) {
245        assert!(input_degree >= self.source.min_degree());
246        assert!(
247            result.as_slice().len() <= self.target.dimension(input_degree - self.degree_shift),
248            "restricted result longer than target dimension"
249        );
250        let mut cursor = self.source.opgen_cursor(input_degree);
251        self.apply_to_basis_element_with(&mut cursor, result, coeff, input_index);
252    }
253
254    /// A truncating variant of [`ModuleHomomorphism::get_partial_matrix`] whose target spans only
255    /// the first `target_dim` basis elements of degree `degree - degree_shift`.
256    ///
257    /// Unlike [`ModuleHomomorphism::get_partial_matrix`], which sizes the matrix from the target's
258    /// dimension, this takes the number of columns from the caller. The support requirement of
259    /// [`Self::apply_to_basis_element_restricted`] applies to every row. See
260    /// [`ModuleHomomorphism::get_matrix`] for the cursor.
261    pub fn get_partial_matrix_restricted(
262        &self,
263        degree: i32,
264        inputs: &[usize],
265        target_dim: usize,
266    ) -> Matrix {
267        assert!(
268            target_dim <= self.target.dimension(degree - self.degree_shift),
269            "restricted result longer than target dimension"
270        );
271        let mut matrix = Matrix::new(self.prime(), inputs.len(), target_dim);
272        if target_dim == 0 {
273            return matrix;
274        }
275
276        let source = &*self.source;
277        matrix.maybe_par_iter_mut().enumerate().for_each_init(
278            || source.opgen_cursor(degree),
279            |cursor, (i, row)| self.apply_to_basis_element_with(cursor, row, 1, inputs[i]),
280        );
281
282        matrix
283    }

285    pub fn differential_density(&self, degree: i32) -> f32 {
286        let outputs = &self.outputs[degree];
287        if outputs.is_empty() {
288            f32::NAN
289        } else {
290            outputs.iter().map(FpVector::density).sum::<f32>() / outputs.len() as f32
291        }
292    }

294    pub fn extend_by_zero(&self, degree: i32) {
295        let p = self.prime();
296        self.outputs.extend(degree, |i| {
297            let num_gens = self.source.number_of_gens_in_degree(i);
298            let dimension = self.target.dimension(i - self.degree_shift);
299            let mut new_outputs: Vec<FpVector> = Vec::with_capacity(num_gens);
300            for _ in 0..num_gens {
301                new_outputs.push(FpVector::new(p, dimension));
302            }
303            new_outputs
304        });
305    }

307    pub fn add_generators_from_big_vector(&self, degree: i32, outputs_vectors: FpSlice) {
308        let p = self.prime();
309        let new_generators = self.source.number_of_gens_in_degree(degree);
310        let target_dimension = self.target.dimension(degree - self.degree_shift);
311        let mut new_outputs: Vec<FpVector> = Vec::with_capacity(new_generators);
312        for _ in 0..new_generators {
313            new_outputs.push(FpVector::new(p, target_dimension));
314        }
315        if target_dimension == 0 {
316            self.outputs.push_checked(new_outputs, degree);
317            return;
318        }
319        for (i, new_output) in new_outputs.iter_mut().enumerate() {
320            new_output
321                .as_slice_mut()
322                .assign(outputs_vectors.restrict(target_dimension * i, target_dimension * (i + 1)));
323        }
324        self.outputs.push_checked(new_outputs, degree);
325    }

327    /// A MatrixSlice will do but there is no application of this struct, so it doesn't exist yet...
328    pub fn add_generators_from_matrix_rows(&self, degree: i32, mut matrix: MatrixSliceMut) {
329        let p = self.prime();
330        let new_generators = self.source.number_of_gens_in_degree(degree);
331        let target_dimension = self.target.dimension(degree - self.degree_shift);
332
333        let mut new_outputs: Vec<FpVector> = Vec::with_capacity(new_generators);
334        for _ in 0..new_generators {
335            new_outputs.push(FpVector::new(p, target_dimension));
336        }
337        if target_dimension == 0 {
338            self.outputs.push_checked(new_outputs, degree);
339            return;
340        }
341        for (i, new_output) in new_outputs.iter_mut().enumerate() {
342            new_output.as_slice_mut().assign(matrix.row(i));
343        }
344        self.outputs.push_checked(new_outputs, degree);
345    }

347    pub fn add_generators_from_rows(&self, degree: i32, rows: Vec<FpVector>) {
348        self.outputs.push_checked(rows, degree);
349    }

398impl<const U: bool, A: MuAlgebra<U>> MuFreeModuleHomomorphism<U, MuFreeModule<U, A>> {
399    /// Given f: M -> N, compute the dual f*: Hom(N, k) -> Hom(M, k) in source (N) degree t.
400    pub fn hom_k(&self, t: i32) -> Vec<Vec<u32>> {
401        let source_dim = self.source.number_of_gens_in_degree(t + self.degree_shift);
402        let target_dim = self.target.number_of_gens_in_degree(t);
403        if target_dim == 0 {
404            return vec![];
405        }
406        let mut result = vec![vec![0; source_dim]; target_dim];
407
408        let offset = self.target.generator_offset(t, t, 0);
409        #[allow(clippy::needless_range_loop)]
410        for i in 0..source_dim {
411            let output = self.output(t + self.degree_shift, i);
412            #[allow(clippy::needless_range_loop)]
413            for j in 0..target_dim {
414                result[j][i] = output.entry(offset + j);
415            }
416        }
417        result
418    }
