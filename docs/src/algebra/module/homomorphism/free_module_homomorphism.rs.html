37impl<const U: bool, M: Module> MuFreeModuleHomomorphism<U, M>
38where
39    M::Algebra: MuAlgebra<U>,
40{
41    /// Apply to one basis element, taking its `(operation, generator)` pair from `cursor`.
42    ///
43    /// The degree is the cursor's, so a caller cannot pair a cursor with a mismatched degree.
44    fn apply_to_basis_element_with(
45        &self,
46        cursor: &mut OpGenCursor<'_, U, M::Algebra>,
47        result: FpSliceMut,
48        coeff: u32,
49        input_index: usize,
50    ) {
51        let input_degree = cursor.degree();
52        let output_degree = input_degree - self.degree_shift;
53        assert_eq!(
54            self.target.dimension(output_degree),
55            result.as_slice().len()
56        );
57        let OperationGeneratorPair {
58            operation_degree,
59            generator_degree,
60            operation_index,
61            generator_index,
62        } = cursor.get(input_index);
63
64        if generator_degree >= self.min_degree() {
65            let output_on_generator = self.output(generator_degree, generator_index);
66            self.target.act(
67                result,
68                coeff,
69                operation_degree,
70                operation_index,
71                generator_degree - self.degree_shift,
72                output_on_generator.as_slice(),
73            );
74        }
75    }
76}

78impl<const U: bool, M: Module> ModuleHomomorphism for MuFreeModuleHomomorphism<U, M>
79where
80    M::Algebra: MuAlgebra<U>,
81{
82    type Source = MuFreeModule<U, M::Algebra>;
83    type Target = M;
84
85    fn source(&self) -> Arc<Self::Source> {
86        Arc::clone(&self.source)
87    }
88
89    fn target(&self) -> Arc<Self::Target> {
90        Arc::clone(&self.target)
91    }
92
93    fn degree_shift(&self) -> i32 {
94        self.degree_shift
95    }
96
97    /// Builds a cursor for this one element. Prefer [`Self::apply`] or [`Self::get_matrix`] when
98    /// applying to several elements of a degree, so that one cursor serves them all.
99    fn apply_to_basis_element(
100        &self,
101        result: FpSliceMut,
102        coeff: u32,
103        input_degree: i32,
104        input_index: usize,
105    ) {
106        assert!(input_degree >= self.source.min_degree());
107        let mut cursor = self.source.opgen_cursor(input_degree);
108        self.apply_to_basis_element_with(&mut cursor, result, coeff, input_index);
109    }
110
111    /// Reuses one cursor across the terms of `input`, which `iter_nonzero` yields in ascending
112    /// order.
113    fn apply(&self, mut result: FpSliceMut, coeff: u32, input_degree: i32, input: FpSlice) {
114        assert!(input_degree >= self.source.min_degree());
115        let p = self.prime();
116        let mut cursor = self.source.opgen_cursor(input_degree);
117        for (i, v) in input.iter_nonzero() {
118            self.apply_to_basis_element_with(&mut cursor, result.copy(), (coeff * v) % p, i);
119        }
120    }
121
122    /// Builds one cursor per worker, which then walks that worker's rows in ascending order.
123    fn get_matrix(&self, mut matrix: MatrixSliceMut, degree: i32) {
124        assert_eq!(self.source.dimension(degree), matrix.rows());
125        assert_eq!(
126            self.target.dimension(degree - self.degree_shift),
127            matrix.columns()
128        );
129
130        if matrix.columns() == 0 {
131            return;
132        }
133
134        let source = &*self.source;
135        matrix.maybe_par_iter_mut().enumerate().for_each_init(
136            || source.opgen_cursor(degree),
137            |cursor, (i, row)| self.apply_to_basis_element_with(cursor, row, 1, i),
138        );
139    }
140
141    /// See [`Self::get_matrix`] for the cursor.
142    ///
143    /// The columns are the target's dimension in the SHIFTED degree, matching what
144    /// [`Self::get_matrix`] asserts and what each row is then filled to.
145    fn get_partial_matrix(&self, degree: i32, inputs: &[usize]) -> Matrix {
146        let mut matrix = Matrix::new(
147            self.prime(),
148            inputs.len(),
149            self.target.dimension(degree - self.degree_shift),
150        );
151
152        if matrix.columns() == 0 {
153            return matrix;
154        }
155
156        let source = &*self.source;
157        matrix.maybe_par_iter_mut().enumerate().for_each_init(
158            || source.opgen_cursor(degree),
159            |cursor, (i, row)| self.apply_to_basis_element_with(cursor, row, 1, inputs[i]),
160        );
161
162        matrix
163    }
164
165    fn quasi_inverse(&self, degree: i32) -> Option<&QuasiInverse> {
166        self.quasi_inverses.get(degree).and_then(Option::as_ref)
167    }
168
169    fn kernel(&self, degree: i32) -> Option<&Subspace> {
170        self.kernels.get(degree).and_then(Option::as_ref)
171    }
172
173    fn image(&self, degree: i32) -> Option<&Subspace> {
174        self.images.get(degree).and_then(Option::as_ref)
175    }
176
177    fn compute_auxiliary_data_through_degree(&self, degree: i32) {
178        self.kernels.extend(degree, |i| {
179            let (image, kernel, qi) = self.auxiliary_data(i);
180            self.images.push_checked(Some(image), i);
181            self.quasi_inverses.push_checked(Some(qi), i);
182            Some(kernel)
183        });
184    }
185}

191    pub fn new(
192        source: Arc<MuFreeModule<U, M::Algebra>>,
193        target: Arc<M>,
194        degree_shift: i32,
195    ) -> Self {
196        let min_degree = std::cmp::max(source.min_degree(), target.min_degree() + degree_shift);
197        let outputs = OnceBiVec::new(min_degree);
198        let kernels = OnceBiVec::new(min_degree);
199        let images = OnceBiVec::new(min_degree);
200        let quasi_inverses = OnceBiVec::new(min_degree);
201        Self {
202            source,
203            target,
204            outputs,
205            images,
206            kernels,
207            quasi_inverses,
208            min_degree,
209            degree_shift,
210        }
211    }
212
213    pub fn degree_shift(&self) -> i32 {
214        self.degree_shift
215    }
216
217    pub fn min_degree(&self) -> i32 {
218        self.min_degree
219    }
220
221    pub fn next_degree(&self) -> i32 {
222        self.outputs.len()
223    }
224
225    pub fn output(&self, generator_degree: i32, generator_index: usize) -> &FpVector {
226        assert!(
227            generator_degree >= self.min_degree(),
228            "generator_degree {} less than min degree {}",
229            generator_degree,
230            self.min_degree()
231        );
232        assert!(
233            generator_index < self.source.number_of_gens_in_degree(generator_degree),
234            "generator_index {} greater than number of generators {}",
235            generator_index,
236            self.source.number_of_gens_in_degree(generator_degree)
237        );
238        &self.outputs[generator_degree][generator_index]
239    }
240
241    pub fn differential_density(&self, degree: i32) -> f32 {
242        let outputs = &self.outputs[degree];
243        if outputs.is_empty() {
244            f32::NAN
245        } else {
246            outputs.iter().map(FpVector::density).sum::<f32>() / outputs.len() as f32
247        }
248    }
249
250    pub fn extend_by_zero(&self, degree: i32) {
251        let p = self.prime();
252        self.outputs.extend(degree, |i| {
253            let num_gens = self.source.number_of_gens_in_degree(i);
254            let dimension = self.target.dimension(i - self.degree_shift);
255            let mut new_outputs: Vec<FpVector> = Vec::with_capacity(num_gens);
256            for _ in 0..num_gens {
257                new_outputs.push(FpVector::new(p, dimension));
258            }
259            new_outputs
260        });
261    }
262
263    pub fn add_generators_from_big_vector(&self, degree: i32, outputs_vectors: FpSlice) {
264        let p = self.prime();
265        let new_generators = self.source.number_of_gens_in_degree(degree);
266        let target_dimension = self.target.dimension(degree - self.degree_shift);
267        let mut new_outputs: Vec<FpVector> = Vec::with_capacity(new_generators);
268        for _ in 0..new_generators {
269            new_outputs.push(FpVector::new(p, target_dimension));
270        }
271        if target_dimension == 0 {
272            self.outputs.push_checked(new_outputs, degree);
273            return;
274        }
275        for (i, new_output) in new_outputs.iter_mut().enumerate() {
276            new_output
277                .as_slice_mut()
278                .assign(outputs_vectors.restrict(target_dimension * i, target_dimension * (i + 1)));
279        }
280        self.outputs.push_checked(new_outputs, degree);
281    }
282
283    /// A MatrixSlice will do but there is no application of this struct, so it doesn't exist yet...
284    pub fn add_generators_from_matrix_rows(&self, degree: i32, mut matrix: MatrixSliceMut) {
285        let p = self.prime();
286        let new_generators = self.source.number_of_gens_in_degree(degree);
287        let target_dimension = self.target.dimension(degree - self.degree_shift);
288
289        let mut new_outputs: Vec<FpVector> = Vec::with_capacity(new_generators);
290        for _ in 0..new_generators {
291            new_outputs.push(FpVector::new(p, target_dimension));
292        }
293        if target_dimension == 0 {
294            self.outputs.push_checked(new_outputs, degree);
295            return;
296        }
297        for (i, new_output) in new_outputs.iter_mut().enumerate() {
298            new_output.as_slice_mut().assign(matrix.row(i));
299        }
300        self.outputs.push_checked(new_outputs, degree);
301    }
302
303    pub fn add_generators_from_rows(&self, degree: i32, rows: Vec<FpVector>) {
304        self.outputs.push_checked(rows, degree);
305    }

307    /// Add the image of a bidegree out of order. See

322    pub fn apply_to_generator(&self, result: &mut FpVector, coeff: u32, degree: i32, idx: usize) {
323        let output_on_gen = self.output(degree, idx);
324        result.add(output_on_gen, coeff);
325    }

354impl<const U: bool, A: MuAlgebra<U>> MuFreeModuleHomomorphism<U, MuFreeModule<U, A>> {
355    /// Given f: M -> N, compute the dual f*: Hom(N, k) -> Hom(M, k) in source (N) degree t.
356    pub fn hom_k(&self, t: i32) -> Vec<Vec<u32>> {
357        let source_dim = self.source.number_of_gens_in_degree(t + self.degree_shift);
358        let target_dim = self.target.number_of_gens_in_degree(t);
359        if target_dim == 0 {
360            return vec![];
361        }
362        let mut result = vec![vec![0; source_dim]; target_dim];
363
364        let offset = self.target.generator_offset(t, t, 0);
365        #[allow(clippy::needless_range_loop)]
366        for i in 0..source_dim {
367            let output = self.output(t + self.degree_shift, i);
368            #[allow(clippy::needless_range_loop)]
369            for j in 0..target_dim {
370                result[j][i] = output.entry(offset + j);
371            }
372        }
373        result
374    }
