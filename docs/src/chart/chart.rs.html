1use algebra::Algebra;
2use ext::{
3    chain_complex::{ChainComplex, FreeChainComplex},
4    utils::query_module,
5};
6use sseq::charting::{SeqSeeBackend, SvgBackend, TikzBackend};
7
8fn main() -> anyhow::Result<()> {
9    ext::utils::init_logging()?;
10
11    let resolution = query_module(None, false)?;
12
13    let format = query::with_default("Output format (svg/tikz/seqsee)", "svg", |x| match x {
14        "svg" | "tikz" | "seqsee" => Ok(x.to_string()),
15        _ => Err(format!(
16            "unknown format '{x}'; expected one of svg, tikz, seqsee"
17        )),
18    });
19
20    let sseq = resolution.to_sseq();
21    let products: Vec<_> = resolution
22        .algebra()
23        .default_filtration_one_products()
24        .into_iter()
25        .map(|(name, op_deg, op_idx)| (name, resolution.filtration_one_products(op_deg, op_idx)))
26        .collect();
27
28    let out = std::io::stdout();
29    match format.as_str() {
30        "svg" => {
31            sseq.write_to_graph(SvgBackend::new(out), 2, false, products.iter(), |_| Ok(()))?
32        }
33        "tikz" => {
34            sseq.write_to_graph(TikzBackend::new(out), 2, false, products.iter(), |_| Ok(()))?
35        }
36        "seqsee" => {
37            sseq.write_to_graph(SeqSeeBackend::new(out), 2, false, products.iter(), |_| {
38                Ok(())
39            })?
40        }
41        _ => unreachable!(),
42    }
43    Ok(())
44}
